83Ocp1Header::Ocp1Header(const ByteVector& memory)
84 : m_syncVal(static_cast<std::uint8_t>(0)),
85 m_protoVers(static_cast<std::uint16_t>(0)),
86 m_msgSize(static_cast<std::uint32_t>(0)),
87 m_msgType(static_cast<std::uint8_t>(0)),
88 m_msgCnt(static_cast<std::uint16_t>(0))
89{
90 assert(memory.size() >= 10); // Not enough data to fit even an Ocp1Header.
91 if (memory.size() >= 10)
92 {
93 m_syncVal = memory.at(0);
94 assert(m_syncVal == 0x3b); // Message does not start with the sync byte.
95
96 m_protoVers = ReadUint16(memory.data() + 1);
97 assert(m_protoVers == 1); // Protocol version is expected to be 1.
98
99 m_msgSize = ReadUint32(memory.data() + 3);
100 assert(m_msgSize >= Ocp1HeaderSize); // Message has unexpected size.
101
102 m_msgType = memory.at(7);
103 assert(m_msgType <= Ocp1Message::KeepAlive); // Message type outside expected range.
104
105 m_msgCnt = ReadUint16(memory.data() + 8);
106 assert(m_msgCnt > 0); // At least one message expected.
107 }
108}
109
110bool Ocp1Header::IsValid() const
111{
112 return ((m_syncVal == 0x3b) && (m_protoVers == 1) && (m_msgSize >= Ocp1HeaderSize) &&
113 (m_msgType <= Ocp1Message::KeepAlive) && (m_msgCnt > 0));
114}

134std::uint32_t Ocp1Header::CalculateMessageSize(std::uint8_t msgType, size_t parameterDataLength)
135{
136 std::uint32_t ret(0);
137
138 switch (msgType)
139 {
140 case Ocp1Message::Command:
141 case Ocp1Message::CommandResponseRequired:
142 ret = static_cast<std::uint32_t>(26 + parameterDataLength);
143 break;
144 case Ocp1Message::Notification:
145 ret = static_cast<std::uint32_t>(37 + parameterDataLength);
146 break;
147 case Ocp1Message::Response:
148 ret = static_cast<std::uint32_t>(19 + parameterDataLength);
149 break;
150 case Ocp1Message::KeepAlive:
151 ret = static_cast<std::uint32_t>(9 + parameterDataLength);
152 break;
153 default:
154 break;
155 }
156
157 return ret;
158}

169std::unique_ptr<Ocp1Message> Ocp1Message::UnmarshalOcp1Message(const ByteVector& receivedData)
170{
171 // Ocp1Header's own constructor asserts on this same condition, which aborts
172 // rather than failing gracefully in a debug build; check first so a short
173 // buffer is rejected instead of crashing the process.
174 if (receivedData.size() < Ocp1Header::Ocp1HeaderSize)
175 return nullptr;
176
177 Ocp1Header header(receivedData);
178 if (!header.IsValid())
179 return nullptr;
180
181 switch (header.GetMessageType())
182 {
183 case Notification:
184 {
185 // Peer-declared msgSize only guarantees Ocp1HeaderSize bytes; guard the fixed fields read below.
186 if (receivedData.size() < 25)
187 return nullptr;
188
189 std::uint32_t notificationSize(ReadUint32(receivedData.data() + 10));
190 std::uint32_t newValueSize = notificationSize - 28;
191 if (newValueSize < 1)
192 return nullptr;
193
194 // Not a valid object number.
195 std::uint32_t targetOno(ReadUint32(receivedData.data() + 14));
196 if (targetOno == 0)
197 return nullptr;
198
199 // Method DefinitionLevel expected to be 3 (OcaSubscriptionManager)
200 std::uint16_t methodDefLevel(ReadUint16(receivedData.data() + 18));
201 if (methodDefLevel < 1)
202 return nullptr;
203
204 // Method index expected to be 1 (AddSubscription)
205 std::uint16_t methodIdx(ReadUint16(receivedData.data() + 20));
206 if (methodIdx < 1)
207 return nullptr;
208
209 // At least one parameter expected.
210 std::uint8_t paramCount = receivedData[22];
211 if (paramCount < 1)
212 return nullptr;
213
214 std::uint16_t contextSize(ReadUint16(receivedData.data() + 23));
215
216 // contextSize is peer-controlled; re-validate before using it as a read offset.
217 if (receivedData.size() < static_cast<std::size_t>(37) + contextSize)
218 return nullptr;
219
220 // Not a valid object number.
221 std::uint32_t emitterOno(ReadUint32(receivedData.data() + 25 + contextSize));
222 if (emitterOno == 0)
223 return nullptr;
224
225 // Event definiton level expected to be 1 (OcaRoot).
226 std::uint16_t eventDefLevel(ReadUint16(receivedData.data() + 29 + contextSize));
227 if (eventDefLevel != 1)
228 return nullptr;
229
230 // Event index expected to be 1 (OCA_EVENT_PROPERTY_CHANGED).
231 std::uint16_t eventIdx(ReadUint16(receivedData.data() + 31 + contextSize));
232 if (eventIdx != 1)
233 return nullptr;
234
235 // Property definition level expected to be > 0.
236 std::uint16_t propDefLevel(ReadUint16(receivedData.data() + 33 + contextSize));
237 if (propDefLevel == 0)
238 return nullptr;
239
240 // Property index expected to be > 0.
241 std::uint16_t propIdx(ReadUint16(receivedData.data() + 35 + contextSize));
242 if (propIdx == 0)
243 return nullptr;
244
245 // notificationSize (and thus newValueSize) is peer-controlled; re-validate before slicing.
246 if (receivedData.size() < static_cast<std::size_t>(37) + contextSize + newValueSize)
247 return nullptr;
248
249 const auto parameterData = ByteVector(receivedData.begin() + 37 + contextSize,
250 receivedData.begin() + 37 + contextSize + newValueSize);
251
252 return std::make_unique<Ocp1Notification>(emitterOno, propDefLevel, propIdx, paramCount, parameterData);
253 }
254
255 case Response:
256 {
257 // Peer-declared msgSize only guarantees Ocp1HeaderSize bytes; guard the fixed fields read below.
258 if (receivedData.size() < 20)
259 return nullptr;
260
261 std::uint32_t responseSize(ReadUint32(receivedData.data() + 10));
262 std::uint32_t parameterDataLength = responseSize - 10;
263 if (responseSize < 10)
264 return nullptr;
265
266 // Not a valid handle.
267 std::uint32_t handle(ReadUint32(receivedData.data() + 14));
268 if (handle == 0)
269 return nullptr;
270
271 const std::uint8_t status = receivedData[18];
272 const std::uint8_t paramCount = receivedData[19];
273
274 // responseSize (and thus parameterDataLength) is peer-controlled; re-validate before slicing.
275 if (receivedData.size() < static_cast<std::size_t>(20) + parameterDataLength)
276 return nullptr;
277
278 const auto parameterData = [&receivedData, &parameterDataLength]() {
279 if (parameterDataLength == 0)
280 return ByteVector{};
281 else
282 return ByteVector(receivedData.begin() + 20, receivedData.begin() + 20 + parameterDataLength);
283 }();
284 assert(parameterData.size() == parameterDataLength);
285
286 return std::make_unique<Ocp1Response>(handle, status, paramCount, parameterData);
287 }
288
289 case KeepAlive:
290 {
291 // Peer-declared msgSize only guarantees Ocp1HeaderSize bytes; guard the read below.
292 if (receivedData.size() < 12)
293 return nullptr;
294
295 std::uint16_t heartbeat(ReadUint16(receivedData.data() + 10));
296
297 return std::make_unique<Ocp1KeepAlive>(heartbeat);
298 }
299
300 case Command:
301 {
302 // Not used in this implementation. See CommandResponseRequired instead.
303 return nullptr;
304 }
305 case CommandResponseRequired:
306 {
307 constexpr std::size_t commandSizeOffset = Ocp1Header::Ocp1HeaderSize;
308 constexpr std::size_t handleOffset = commandSizeOffset + 4;
309 constexpr std::size_t targetOnoOffset = handleOffset + 4;
310 constexpr std::size_t methodDefLevelOffset = targetOnoOffset + 4;
311 constexpr std::size_t methodIdxOffset = methodDefLevelOffset + 2;
312 constexpr std::size_t paramCountOffset = methodIdxOffset + 2;
313 constexpr std::uint32_t minimumCommandSize = 17; // Size without parameters
314 constexpr std::size_t parameterDataOffset = paramCountOffset + 1;
315
316 // Guard against reading commandSize itself past the end of a truncated frame.
317 if (receivedData.size() < commandSizeOffset + sizeof(std::uint32_t))
318 return nullptr;
319
320 // commandSize(4) + handle(4) + targetOno(4) + methodDefLevel(2) + methodIdx(2) + paramCount(1) = 17 bytes
321 const std::uint32_t commandSize(ReadUint32(receivedData.data() + commandSizeOffset));
322 if (commandSize < minimumCommandSize)
323 return nullptr;
324
325 if(receivedData.size() < Ocp1Header::Ocp1HeaderSize + commandSize)
326 return nullptr;
327
328 std::uint32_t handle(ReadUint32(receivedData.data() + handleOffset));
329 const bool isInvalidHandle = (handle == 0);
330 if (isInvalidHandle)
331 return nullptr;
332
333 const std::uint32_t targetOno(ReadUint32(receivedData.data() + targetOnoOffset));
334 const bool isInvalidTargetOno = (targetOno == 0);
335 if (isInvalidTargetOno)
336 return nullptr;
337
338 const std::uint16_t methodDefLevel(ReadUint16(receivedData.data() + methodDefLevelOffset));
339 const bool isInvalidMethodDefLevel = (methodDefLevel < 1);
340 if (isInvalidMethodDefLevel)
341 return nullptr;
342
343 const std::uint16_t methodIdx(ReadUint16(receivedData.data() + methodIdxOffset));
344 const bool isInvalidMethodIdx = (methodIdx < 1);
345 if (isInvalidMethodIdx)
346 return nullptr;
347
348 const std::uint8_t paramCount = receivedData[paramCountOffset];
349
350 const auto parameterData = [&receivedData, &paramCount, &parameterDataOffset]() {
351 if (paramCount == 0)
352 return ByteVector{};
353 else if(receivedData.begin() + parameterDataOffset >= receivedData.end())
354 return ByteVector{};
355 else
356 return ByteVector(receivedData.begin() + parameterDataOffset, receivedData.end());
357 }();
358
359 auto result = std::make_unique<Ocp1CommandResponseRequired>(targetOno, methodDefLevel, methodIdx, paramCount, parameterData);
360 result->SetHandle(handle);
361 return result;
362 }
363 default:
364 return nullptr;
365 }
366}
