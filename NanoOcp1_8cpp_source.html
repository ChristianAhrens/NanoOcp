150void NanoOcp1Client::timerCallback()
151{
152 // Do NOT call stopTimer() here on success. connectToSocket() already routes
153 // through initialise() -> connectionMadeInt() -> connectionMade(), and
154 // connectionMade() itself calls stopTimer(). When callbacksOnMessageThread
155 // is true (the default), that call happens asynchronously on the dispatcher
156 // thread — concurrently with this very callback still running on the timer
157 // thread. Two concurrent stopTimer() calls on the same NanoTimer deadlock:
158 // the foreign-thread call blocks in std::thread::join() waiting for this
159 // timer thread's callback to return, while this thread's own reentrant
160 // stopTimer() call (guarded to skip join() on self) still has to wait for
161 // the same m_lifecycleMutex the joining thread is holding — which it can
162 // never release until this thread returns. So leave stopping the timer
163 // solely to connectionMade(), and just avoid redialling an already-live
164 // connection while that callback is in flight.
165 if (!isConnected())
166 connectToSocket(getAddress(), getPort(), 50);
167}
