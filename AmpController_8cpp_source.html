46void AmpController::setAmpType(AmpType type, std::uint16_t channelCount)
47{
48 assert(getState() == State::Disconnected);
49 m_ampType = type;
50 m_channelCount = channelCount;
51 rebuildTrackedObjects();
52}

54void AmpController::rebuildTrackedObjects()
55{
56 clearTrackedObjects();
57
58 // ── Power (global, no channel index) ─────────────────────────────────────
59 switch (m_ampType)
60 {
61 case AmpType::FiveD:
62 trackObject(
63 std::make_unique<Amp5D::dbOcaObjectDef_Settings_PwrOn>(),
64 [this](const ByteVector& data) {
65 if (onPower) onPower(DataToUint16(data) > 0);
66 });
67 break;
68 default: // Dx and Dy share the same PwrOn definition
69 trackObject(
70 std::make_unique<AmpDxDy::dbOcaObjectDef_Settings_PwrOn>(),
71 [this](const ByteVector& data) {
72 if (onPower) onPower(DataToUint16(data) > 0);
73 });
74 break;
75 }
76
77 // ── Per-channel objects ───────────────────────────────────────────────────
78 for (std::uint16_t ch = 1; ch <= m_channelCount; ++ch)
79 {
80 // Gain — common to all amp types
81 trackObject(
82 std::make_unique<AmpGeneric::dbOcaObjectDef_Config_PotiLevel>(ch),
83 [this, ch](const ByteVector& data) {
84 if (onChannelGain) onChannelGain(ch, DataToFloat(data));
85 });
86
87 // Mute — common to all amp types.
88 // Wire encoding from the device: 1 = muted, 2 = unmuted.
89 trackObject(
90 std::make_unique<AmpGeneric::dbOcaObjectDef_Config_Mute>(ch),
91 [this, ch](const ByteVector& data) {
92 if (onChannelMute) onChannelMute(ch, DataToUint8(data) == 1);
93 });
94
95 // ISP, GR, OVL — Dx/Dy share definitions; 5D uses different ONos
96 switch (m_ampType)
97 {
98 case AmpType::FiveD:
99 trackObject(
100 std::make_unique<Amp5D::dbOcaObjectDef_ChStatus_Isp>(ch),
101 [this, ch](const ByteVector& data) {
102 if (onChannelISP) onChannelISP(ch, DataToBool(data));
103 });
104 trackObject(
105 std::make_unique<Amp5D::dbOcaObjectDef_ChStatus_Gr>(ch),
106 [this, ch](const ByteVector& data) {
107 if (onChannelGR) onChannelGR(ch, DataToBool(data));
108 });
109 trackObject(
110 std::make_unique<Amp5D::dbOcaObjectDef_ChStatus_Ovl>(ch),
111 [this, ch](const ByteVector& data) {
112 if (onChannelOVL) onChannelOVL(ch, DataToBool(data));
113 });
114 break;
115 default: // Dx and Dy
116 trackObject(
117 std::make_unique<AmpDxDy::dbOcaObjectDef_ChStatus_Isp>(ch),
118 [this, ch](const ByteVector& data) {
119 if (onChannelISP) onChannelISP(ch, DataToBool(data));
120 });
121 trackObject(
122 std::make_unique<AmpDxDy::dbOcaObjectDef_ChStatus_Gr>(ch),
123 [this, ch](const ByteVector& data) {
124 if (onChannelGR) onChannelGR(ch, DataToBool(data));
125 });
126 trackObject(
127 std::make_unique<AmpDxDy::dbOcaObjectDef_ChStatus_Ovl>(ch),
128 [this, ch](const ByteVector& data) {
129 if (onChannelOVL) onChannelOVL(ch, DataToBool(data));
130 });
131 break;
132 }
133
134 // GrHead (headroom) — all three amp types have distinct ONos
135 switch (m_ampType)
136 {
137 case AmpType::Dx:
138 trackObject(
139 std::make_unique<AmpDx::dbOcaObjectDef_ChStatus_GrHead>(ch),
140 [this, ch](const ByteVector& data) {
141 if (onChannelHeadroom) onChannelHeadroom(ch, DataToFloat(data));
142 });
143 break;
144 case AmpType::Dy:
145 trackObject(
146 std::make_unique<AmpDy::dbOcaObjectDef_ChStatus_GrHead>(ch),
147 [this, ch](const ByteVector& data) {
148 if (onChannelHeadroom) onChannelHeadroom(ch, DataToFloat(data));
149 });
150 break;
151 case AmpType::FiveD:
152 trackObject(
153 std::make_unique<Amp5D::dbOcaObjectDef_ChStatus_GrHead>(ch),
154 [this, ch](const ByteVector& data) {
155 if (onChannelHeadroom) onChannelHeadroom(ch, DataToFloat(data));
156 });
157 break;
158 }
159 }
160}

165bool AmpController::setPower(bool on)
166{
167 const Variant v(static_cast<std::uint16_t>(on ? 1 : 0));
168 switch (m_ampType)
169 {
170 case AmpType::FiveD:
171 {
172 Amp5D::dbOcaObjectDef_Settings_PwrOn def;
173 return setValue(def, v);
174 }
175 default: // Dx and Dy
176 {
177 AmpDxDy::dbOcaObjectDef_Settings_PwrOn def;
178 return setValue(def, v);
179 }
180 }
181}
182
183bool AmpController::setChannelGain(std::uint16_t channel, float gainDb)
184{
185 AmpGeneric::dbOcaObjectDef_Config_PotiLevel def(channel);
186 return setValue(def, Variant(gainDb));
187}
188
189bool AmpController::setChannelMute(std::uint16_t channel, bool mute)
190{
191 AmpGeneric::dbOcaObjectDef_Config_Mute def(channel);
192 // d&b amp convention: 1 = muted, 2 = unmuted (not the boolean 0/1)
193 return setValue(def, Variant(static_cast<std::uint8_t>(mute ? 1 : 2)));
194}
