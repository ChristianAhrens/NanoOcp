40// ── Panel dimensions ──────────────────────────────────────────────────────────
41// Amp and Soundscape-overview modes use exactly 14 fixed rows + 5 log rows = 19
42// total, never resized.
43// Layout (fixed rows):
44// 1 title 2 sep 3 host/port
45// 4 status 5 mode row 1 6-9 mode rows 2-5
46// 10 sep 11-13 commands 14 Events header
47static constexpr int kLogLines = 5;
48static constexpr int kFixedLines = 14;
49static constexpr int kPanelLines = kFixedLines + kLogLines; // 19
50
51// Soundscape-focus mode instead fills the whole terminal height. It has its own,
52// shorter, fixed-row count; the remaining terminal height becomes the log.
53// 1 title 2 sep 3 host/port 4 status 5 value
54// 6 sep 7-8 commands 9 sep 10 Events header
55static constexpr int kFocusFixedLines = 10;
56static constexpr int kFocusMinLogLines = 3;

134static void pushLog(const std::string& msg)
135{
136 {
137 std::lock_guard<std::mutex> lk(g_stateMutex);
138 g_log.push_back(msg);
139 const auto cap = static_cast<std::size_t>(std::max(1, g_logCapacity.load()));
140 while (g_log.size() > cap)
141 g_log.pop_front();
142 }
143 g_needsRedraw = true;
144}

146// Returns "HH:MM:SS.mmm" for the current local time — used to timestamp entries
147// in Soundscape-focus mode's monitor log.
148static std::string timestampNow()
149{
150 using namespace std::chrono;
151 const auto now = system_clock::now();
152 const auto t = system_clock::to_time_t(now);
153 const auto ms = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
154 std::tm tmBuf{};
155#if defined(_WIN32) || defined(_WIN64)
156 localtime_s(&tmBuf, &t);
157#else
158 localtime_r(&t, &tmBuf);
159#endif
160 std::ostringstream oss;
161 oss << std::put_time(&tmBuf, "%H:%M:%S") << '.' << std::setw(3) << std::setfill('0') << ms.count();
162 return oss.str();
163}

165// Like pushLog(), but prefixes a timestamp — used for Soundscape-focus mode's
166// monitor log so every value update/command is traceable in time.
167static void pushFocusLog(const std::string& msg)
168{
169 pushLog(timestampNow() + " " + msg);
170}
