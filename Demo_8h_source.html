21// ── Demo controller ───────────────────────────────────────────────────────────
22// Wraps AmpController / SoundscapeController, owns the Config for the mode
23// selected on the command line, and exposes one cmd*() method per interactive
24// command. All cmd*()/connect*() methods update g_state and push to the log;
25// they never touch the terminal directly (see Panels.h for that).

40public:
41 struct Config
42 {
43 std::string host{"127.0.0.1"};
44 int port{50014};
45 DemoMode mode{DemoMode::Amp};
46 // Amp-specific
47 NanoOcp1::AmpController::AmpType ampType{NanoOcp1::AmpController::AmpType::Dy};
48 std::uint16_t channelCount{4};
49 // DS100-specific
50 int soundObject{1};
51 // Soundscape-focus-specific (only used when mode == SoundscapeFocus)
52 SORemObjIdent focusParamId{SORemObjIdent::Invalid};
53 std::string focusParamName;
54 int addr2{0};
55 };

57 explicit Demo(Config cfg) : m_cfg(std::move(cfg))
58 {
59 std::lock_guard<std::mutex> lk(g_stateMutex);
60 g_state.address = m_cfg.host;
61 g_state.port = m_cfg.port;
62 g_state.mode = m_cfg.mode;
63 g_state.amp.type = m_cfg.ampType;
64 g_state.amp.channelCount = static_cast<int>(m_cfg.channelCount);
65 g_state.ds100.soundObject = m_cfg.soundObject;
66 g_state.focus.paramName = m_cfg.focusParamName;
67 g_state.focus.paramId = m_cfg.focusParamId;
68 g_state.focus.addr = SORemObjAddr{static_cast<std::int16_t>(m_cfg.soundObject),
69 static_cast<std::int16_t>(m_cfg.addr2)};
70 }

72 ~Demo() { teardown(); }

76 void connect()
77 {
78 teardown();
79
80 switch (m_cfg.mode)
81 {
82 case DemoMode::Amp: connectAmp(); break;
83 case DemoMode::Soundscape: connectDS100(); break;
84 case DemoMode::SoundscapeFocus: connectFocus(); break;
85 }
86 }
87
88 void disconnect()
89 {
90 teardown();
91 {
92 std::lock_guard<std::mutex> lk(g_stateMutex);
93 g_state.ctrlState = CtrlState::Disconnected;
94 switch (m_cfg.mode)
95 {
96 case DemoMode::Amp: resetAmpDisplay(); break;
97 case DemoMode::Soundscape: resetSoDisplay(); break;
98 case DemoMode::SoundscapeFocus: resetFocusDisplay(); break;
99 }
100 }
101 g_needsRedraw = true;
102 if (m_cfg.mode == DemoMode::SoundscapeFocus)
103 pushFocusLog("Disconnected.");
104 else
105 pushLog("Disconnected.");
106 }

306 // ── Soundscape-focus command ──────────────────────────────────────────────
307
308 // Parses `rawValue` according to the focused parameter's last-known data type
309 // and sends it as a SetValue. Requires at least one value to have already
310 // been received, since that is how the demo learns the parameter's type.
311 void cmdSetFocusValue(const std::string& rawValue)
312 {
313 if (!m_ds100) { pushFocusLog("Not connected."); return; }
314
315 NanoOcp1::Ocp1DataType hint = NanoOcp1::OCP1DATATYPE_NONE;
316 SORemObjAddr addr;
317 {
318 std::lock_guard<std::mutex> lk(g_stateMutex);
319 if (g_state.focus.valueKnown)
320 hint = g_state.focus.lastValue.GetDataType();
321 addr = g_state.focus.addr;
322 }
323
324 NanoOcp1::Variant value;
325 std::string err;
326 if (!parseVariantInput(rawValue, hint, value, err))
327 {
328 pushFocusLog("Invalid value: " + err);
329 return;
330 }
331
332 using RO = NanoOcp1::SoundscapeController::RemoteObject;
333 if (!m_ds100->setObjectValue(RO{m_cfg.focusParamId, addr, value}))
334 pushFocusLog("setObjectValue(" + m_cfg.focusParamName + ") failed (not connected?)");
335 else
336 pushFocusLog("Sent " + m_cfg.focusParamName + " -> " + formatVariant(value));
337 }

342 void connectAmp()
343 {
344 m_amp = std::make_unique<NanoOcp1::AmpController>();
345 m_amp->setAmpType(m_cfg.ampType, m_cfg.channelCount);
346
347 m_amp->onStateChanged = [this](CtrlState s) {
348 {
349 std::lock_guard<std::mutex> lk(g_stateMutex);
350 g_state.ctrlState = s;
351 if (s == CtrlState::Disconnected)
352 resetAmpDisplay();
353 }
354 g_needsRedraw = true;
355 pushLog("State: " + stateToStr(s));
356 };
357
358 m_amp->onPower = [](bool on) {
359 {
360 std::lock_guard<std::mutex> lk(g_stateMutex);
361 g_state.amp.powerKnown = true;
362 g_state.amp.powerOn = on;
363 }
364 g_needsRedraw = true;
365 pushLog(std::string("Power -> ") + (on ? "ON" : "OFF"));
366 };
367
368 m_amp->onChannelGain = [](std::uint16_t ch, float dB) {
369 if (ch < 1 || ch > 4) return;
370 {
371 std::lock_guard<std::mutex> lk(g_stateMutex);
372 g_state.amp.ch[ch - 1].gainKnown = true;
373 g_state.amp.ch[ch - 1].gainDb = dB;
374 }
375 g_needsRedraw = true;
376 std::ostringstream oss;
377 oss << "Ch" << ch << " gain -> "
378 << std::fixed << std::setprecision(1) << dB << " dB";
379 pushLog(oss.str());
380 };
381
382 m_amp->onChannelMute = [](std::uint16_t ch, bool muted) {
383 if (ch < 1 || ch > 4) return;
384 {
385 std::lock_guard<std::mutex> lk(g_stateMutex);
386 g_state.amp.ch[ch - 1].muteKnown = true;
387 g_state.amp.ch[ch - 1].muted = muted;
388 }
389 g_needsRedraw = true;
390 pushLog(std::string("Ch") + std::to_string(ch)
391 + " mute -> " + (muted ? "ON" : "off"));
392 };
393
394 m_amp->onChannelISP = [](std::uint16_t ch, bool active) {
395 if (ch < 1 || ch > 4) return;
396 {
397 std::lock_guard<std::mutex> lk(g_stateMutex);
398 g_state.amp.ch[ch - 1].ispKnown = true;
399 g_state.amp.ch[ch - 1].isp = active;
400 }
401 g_needsRedraw = true;
402 if (active)
403 pushLog(std::string("Ch") + std::to_string(ch) + " ISP active");
404 };
405
406 m_amp->onChannelGR = [](std::uint16_t ch, bool active) {

408 {
409 std::lock_guard<std::mutex> lk(g_stateMutex);
410 g_state.amp.ch[ch - 1].grKnown = true;
411 g_state.amp.ch[ch - 1].gr = active;
412 }
413 g_needsRedraw = true;
414 if (active)
415 pushLog(std::string("Ch") + std::to_string(ch) + " GR active");
416 };
417
418 m_amp->onChannelOVL = [](std::uint16_t ch, bool active) {
419 if (ch < 1 || ch > 4) return;
420 {
421 std::lock_guard<std::mutex> lk(g_stateMutex);
422 g_state.amp.ch[ch - 1].ovlKnown = true;
423 g_state.amp.ch[ch - 1].ovl = active;

444 // ── DS100 connection ──────────────────────────────────────────────────────
445
446 void connectDS100()
447 {
448 using ROI = NanoOcp1::SoundscapeController::RemoteObject::RemObjIdent;
449 using ROA = NanoOcp1::SoundscapeController::RemObjAddr;
450 using RO = NanoOcp1::SoundscapeController::RemoteObject;
451
452 m_ds100 = std::make_unique<NanoOcp1::SoundscapeController>();
453
454 const auto so = static_cast<std::int16_t>(m_cfg.soundObject);
455
456 m_ds100->setActiveRemoteObjects({
457 RO{ ROI::MatrixInput_LevelMeterPreMute, ROA{so, 0} },
458 RO{ ROI::MatrixInput_Gain, ROA{so, 0} },
459 RO{ ROI::MatrixInput_Mute, ROA{so, 0} },
460 RO{ ROI::Positioning_SourcePosition, ROA{so, 0} },
461 RO{ ROI::Positioning_SourceSpread, ROA{so, 0} },
462 RO{ ROI::Positioning_SourceDelayMode, ROA{so, 0} },
463 RO{ ROI::MatrixInput_ReverbSendGain, ROA{so, 0} },
464 });
465
466 m_ds100->onStateChanged = [this](CtrlState s) {
467 std::string devModel;
468 if (s == CtrlState::Connected)
469 {
470 devModel = modelStr(m_ds100->getConnectedDeviceModel());
471 const int stack = m_ds100->getOcaStackIdent();
472 if (stack >= 0)
473 devModel += " stack " + std::to_string(stack);
474 }
475 {
476 std::lock_guard<std::mutex> lk(g_stateMutex);
477 g_state.ctrlState = s;
478 g_state.ds100.deviceModel = devModel;
479 if (s == CtrlState::Disconnected)
480 resetSoDisplay();
481 }
482 g_needsRedraw = true;
483 pushLog("State: " + stateToStr(s));
484 };
485
486 m_ds100->onRemoteObjectReceived = [this](const NanoOcp1::SoundscapeController::RemoteObject& ro) -> bool {
487 using Id = NanoOcp1::SoundscapeController::RemoteObject::RemObjIdent;
488 bool ok = false;
489
490 switch (ro.Id)
491 {
492 case Id::MatrixInput_LevelMeterPreMute:
493 {
494 const float dB = ro.Var.ToFloat(&ok);
495 if (ok)
496 {
497 std::lock_guard<std::mutex> lk(g_stateMutex);
498 g_state.ds100.levelKnown = true;
499 g_state.ds100.levelDb = dB;
500 g_needsRedraw = true;
501 }
502 break;
503 }
504 case Id::MatrixInput_Gain:
505 {
506 const float dB = ro.Var.ToFloat(&ok);
507 if (ok)
508 {
509 std::lock_guard<std::mutex> lk(g_stateMutex);
510 g_state.ds100.gainKnown = true;
511 g_state.ds100.gainDb = dB;
512 g_needsRedraw = true;
513 }
514 break;
515 }
516 case Id::MatrixInput_Mute:
517 {
518 const std::uint8_t v = ro.Var.ToUInt8(&ok);
519 if (ok)
520 {
521 std::lock_guard<std::mutex> lk(g_stateMutex);
522 g_state.ds100.muteKnown = true;
523 g_state.ds100.muted = (v == 1);
524 g_needsRedraw = true;
525 }
526 break;
527 }
528 case Id::Positioning_SourcePosition:
529 {
530 const auto xyz = ro.Var.ToPosition(&ok);
531 if (ok)
532 {
533 {
534 std::lock_guard<std::mutex> lk(g_stateMutex);
535 g_state.ds100.posKnown = true;
536 g_state.ds100.posX = xyz[0];
537 g_state.ds100.posY = xyz[1];
538 g_state.ds100.posZ = xyz[2];
539 g_needsRedraw = true;
540 }
541 std::ostringstream oss;
542 oss << "Pos X=" << std::fixed << std::setprecision(3) << xyz[0]
543 << " Y=" << xyz[1] << " Z=" << xyz[2];
544 pushLog(oss.str());
545 }
546 break;
547 }
548 case Id::Positioning_SourceSpread:
549 {
550 const float s = ro.Var.ToFloat(&ok);
551 if (ok)
552 {
553 std::lock_guard<std::mutex> lk(g_stateMutex);
554 g_state.ds100.spreadKnown = true;
555 g_state.ds100.spread = s;
556 g_needsRedraw = true;
557 }
558 break;
559 }
560 case Id::Positioning_SourceDelayMode:
561 {
562 const std::uint8_t dm = ro.Var.ToUInt8(&ok);
563 if (ok)
564 {
565 std::lock_guard<std::mutex> lk(g_stateMutex);
566 g_state.ds100.dmKnown = true;
567 g_state.ds100.delayMode = static_cast<int>(dm);
568 g_needsRedraw = true;
569 }
570 break;
571 }
572 case Id::MatrixInput_ReverbSendGain:
573 {
574 const float dB = ro.Var.ToFloat(&ok);
575 if (ok)
576 {
577 std::lock_guard<std::mutex> lk(g_stateMutex);
578 g_state.ds100.esKnown = true;
579 g_state.ds100.enspaceDb = dB;
580 g_needsRedraw = true;
581 }
582 break;
583 }
584 default:
585 return false;
586 }
587 return ok;
588 };
589
590 pushLog("Connecting to " + m_cfg.host + ":" + std::to_string(m_cfg.port)
591 + " (Soundscape, SO#" + std::to_string(m_cfg.soundObject) + ")...");
592 m_ds100->connect(m_cfg.host, m_cfg.port);
593 }

595 // ── Soundscape-focus connection ───────────────────────────────────────────
596
597 // Subscribes to exactly one remote object and streams every GetValue response
598 // / change notification for it into the timestamped focus log.
599 void connectFocus()
600 {
601 using RO = NanoOcp1::SoundscapeController::RemoteObject;
602
603 m_ds100 = std::make_unique<NanoOcp1::SoundscapeController>();
604
605 const SORemObjAddr addr{static_cast<std::int16_t>(m_cfg.soundObject),
606 static_cast<std::int16_t>(m_cfg.addr2)};
607
608 m_ds100->setActiveRemoteObjects({ RO{ m_cfg.focusParamId, addr } });
609
610 m_ds100->onStateChanged = [this](CtrlState s) {
611 std::string devModel;
612 if (s == CtrlState::Connected)
613 {
614 devModel = modelStr(m_ds100->getConnectedDeviceModel());
615 const int stack = m_ds100->getOcaStackIdent();
616 if (stack >= 0)
617 devModel += " stack " + std::to_string(stack);
618 }
619 {
620 std::lock_guard<std::mutex> lk(g_stateMutex);
621 g_state.ctrlState = s;
622 g_state.ds100.deviceModel = devModel;
623 if (s == CtrlState::Disconnected)
624 resetFocusDisplay();
625 }
626 g_needsRedraw = true;
627 pushFocusLog("State: " + stateToStr(s));
628 };
629
630 m_ds100->onRemoteObjectReceived = [this](const NanoOcp1::SoundscapeController::RemoteObject& ro) -> bool {
631 if (ro.Id != m_cfg.focusParamId)
632 return false;
633 {
634 std::lock_guard<std::mutex> lk(g_stateMutex);
635 g_state.focus.valueKnown = true;
636 g_state.focus.lastValue = ro.Var;
637 ++g_state.focus.updateCount;
638 g_needsRedraw = true;
639 }
640 pushFocusLog(m_cfg.focusParamName + " -> " + formatVariant(ro.Var));
641 return true;
642 };
643
644 std::string addrStr = std::to_string(addr.pri);
645 if (addr.sec != 0)
