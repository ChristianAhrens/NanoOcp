79
80void SoundscapeController::afterConnected()
81{
82 // Reset per-connection device state.
83 m_deviceGuid = "";
84 m_stackIdent = -1;
85 m_connectedModel = DbDeviceModel::Invalid;
86
87 // Query Fixed_GUID first. Subscriptions follow in onUntrackedGetValueResponse()
88 // once we know which OCA revision the firmware supports.
89 DS100::dbOcaObjectDef_Fixed_GUID guidDef;
90 queryObjectValue(guidDef);
91}
92
93void SoundscapeController::onUntrackedGetValueResponse(std::uint32_t ono,
94 const ByteVector& paramData)
95{
96 DS100::dbOcaObjectDef_Fixed_GUID guidDef;
97 if (ono != guidDef.m_targetOno)
98 return;
99
100 bool ok = false;
101 auto guid = DataToString(paramData, &ok);
102 if (ok)
103 processGuidAndSubscribe(guid);
104}

107// ── GUID processing ───────────────────────────────────────────────────────────
108
120void SoundscapeController::processGuidAndSubscribe(const std::string& guid)
121{
122 if (guid == m_deviceGuid) // already handled this GUID for the current connection
123 return;
124
125 if (!setOcaRevisionAndDeviceModel(guid))
126 return;
127 m_deviceGuid = guid;
128
129 // Patch the speaker-position map entries to match the detected OCA revision.
130 RemObjAddr roa;
131 if (m_stackIdent >= 1)
132 {
133 for (std::uint16_t ch = 1; ch <= m_activeOutputChannelCount; ++ch)
134 {
135 roa.pri = ch;
136 m_ROIsToDefsMap[RemoteObject::Positioning_SpeakerPosition][roa]
137 = DS100::dbOcaObjectDef_Positioning_Speaker_Position(ch);
138 m_ROIsToDefsMap[RemoteObject::Positioning_SpeakerGroup][roa]
139 = DS100::dbOcaObjectDef_Positioning_Speaker_Group(ch);
140 }
141 }
142 else
143 {
144 for (std::uint16_t ch = 1; ch <= m_activeOutputChannelCount; ++ch)
145 {
146 roa.pri = ch;
147 m_ROIsToDefsMap[RemoteObject::Positioning_SpeakerPosition][roa]
148 = DS100::dbOcaObjectDef_Positioning_Source_Speaker_Position(ch);
149 }
150 }
151
152 // Rebuild the reverse-lookup map to include the patched ONos.
153 m_ONoToROIMap.clear();
154 for (auto& roiKV : m_ROIsToDefsMap)
155 for (auto& defKV : roiKV.second)
156 m_ONoToROIMap[defKV.second.m_targetOno] = { roiKV.first, defKV.first };
157
158 // Rebuild tracked objects so they reference the correct definitions.
159 rebuildTrackedObjects();
160
161 createObjectSubscriptions();
162 queryObjectValues();
163}

175bool SoundscapeController::setOcaRevisionAndDeviceModel(const std::string& guid)
176{
177 if (guid.size() != 8)
178 return false;
179 if (guid.substr(0, 4) != "DB00")
180 return false;
181
182 DbDeviceModel model;
183 const auto suffix = guid.substr(6, 2);
184 if (suffix == "D0") model = DbDeviceModel::DS100;
185 else if (suffix == "D1") model = DbDeviceModel::DS110;
186 else if (suffix == "D2") model = DbDeviceModel::DS100M;
187 else if (suffix == "DA") model = DbDeviceModel::vCore;
188 else return false;
189
190 const auto versionChars = guid.substr(4, 2);
191 int stackIdent = 0;
192 switch (model)
193 {
194 case DbDeviceModel::DS100:
195 stackIdent = (versionChars >= "0C") ? 1 : 0;
196 break;
197 case DbDeviceModel::DS110:
198 stackIdent = 1; // pre-release firmware without scalability was never released
199 break;
200 case DbDeviceModel::DS100M:
201 stackIdent = (versionChars >= "02") ? 1 : 0;
202 break;
203 case DbDeviceModel::vCore:
204 stackIdent = 1;
205 break;
206 default:
207 return false;
208 }
209
210 m_connectedModel = model;
211 m_stackIdent = stackIdent;
212 return true;
213}

216// ── Tracked-object management ─────────────────────────────────────────────────
217
231void SoundscapeController::rebuildTrackedObjects()
232{
233 clearTrackedObjects();
234
235 for (const auto& obj : m_activeRemoteObjects)
236 {
237 auto roiIt = m_ROIsToDefsMap.find(obj.Id);
238 if (roiIt == m_ROIsToDefsMap.end())
239 continue;
240 auto defIt = roiIt->second.find(obj.Addr);
241 if (defIt == roiIt->second.end())
242 continue;
243
244 const RemoteObject::RemObjIdent roi = obj.Id;
245 const RemObjAddr addr = obj.Addr;
246 const Ocp1DataType dt = dataTypeForRoi(roi);
247
248 auto defCopy = std::unique_ptr<Ocp1CommandDefinition>(defIt->second.Clone());
249
250 trackObject(std::move(defCopy), [this, roi, addr, dt](const ByteVector& data) {
251 Variant val(data, dt);
252 RemoteObject ro(roi, addr, std::move(val));
253 if (onRemoteObjectReceived)
254 onRemoteObjectReceived(ro);
255 });
256 }
257}

520// ── Full ONo map construction ─────────────────────────────────────────────────
521
529void SoundscapeController::createKnownONosMap()
530{
531 using namespace DS100;
532
533 m_ROIsToDefsMap.clear();
534 m_ONoToROIMap.clear();
535
536 // ── Unindexed objects (no channel/record) ────────────────────────────────
537 m_ROIsToDefsMap[RemoteObject::Fixed_GUID][RemObjAddr()] = dbOcaObjectDef_Fixed_GUID();
538 m_ROIsToDefsMap[RemoteObject::Settings_DeviceName][RemObjAddr()] = dbOcaObjectDef_Settings_DeviceName();
539 m_ROIsToDefsMap[RemoteObject::Status_StatusText][RemObjAddr()] = dbOcaObjectDef_Status_StatusText();
540 m_ROIsToDefsMap[RemoteObject::Status_AudioNetworkSampleStatus][RemObjAddr()] = dbOcaObjectDef_Status_AudioNetworkSampleStatus();
541 m_ROIsToDefsMap[RemoteObject::Error_GnrlErr][RemObjAddr()] = dbOcaObjectDef_Error_GnrlErr();
542 m_ROIsToDefsMap[RemoteObject::Error_ErrorText][RemObjAddr()] = dbOcaObjectDef_Error_ErrorText();
543 m_ROIsToDefsMap[RemoteObject::MatrixSettings_ReverbRoomId][RemObjAddr()] = dbOcaObjectDef_MatrixSettings_ReverbRoomId();
544 m_ROIsToDefsMap[RemoteObject::MatrixSettings_ReverbPredelayFactor][RemObjAddr()] = dbOcaObjectDef_MatrixSettings_ReverbPredelayFactor();
545 m_ROIsToDefsMap[RemoteObject::MatrixSettings_ReverbRearLevel][RemObjAddr()] = dbOcaObjectDef_MatrixSettings_ReverbRearLevel();
546 m_ROIsToDefsMap[RemoteObject::Scene_SceneIndex][RemObjAddr()] = dbOcaObjectDef_Scene_SceneIndex();
547 m_ROIsToDefsMap[RemoteObject::Scene_SceneName][RemObjAddr()] = dbOcaObjectDef_Scene_SceneName();
548 m_ROIsToDefsMap[RemoteObject::Scene_SceneComment][RemObjAddr()] = dbOcaObjectDef_Scene_SceneComment();
549
550 // ── Per-input-channel (sound objects) ────────────────────────────────────
551 for (std::int16_t ch = 1; ch <= m_activeInputChannelCount; ++ch)
552 {
553 RemObjAddr roa(ch, RemObjAddr::sc_INV);
554
555 m_ROIsToDefsMap[RemoteObject::Positioning_SourcePosition][roa] = dbOcaObjectDef_Positioning_Source_Position(ch);
556 m_ROIsToDefsMap[RemoteObject::Positioning_SourceSpread][roa] = dbOcaObjectDef_Positioning_Source_Spread(ch);
557 m_ROIsToDefsMap[RemoteObject::Positioning_SourceDelayMode][roa] = dbOcaObjectDef_Positioning_Source_DelayMode(ch);
558 m_ROIsToDefsMap[RemoteObject::Positioning_SourceEnable][roa] = dbOcaObjectDef_Positioning_Source_Enable(ch);
559 m_ROIsToDefsMap[RemoteObject::MatrixInput_Mute][roa] = dbOcaObjectDef_MatrixInput_Mute(ch);
560 m_ROIsToDefsMap[RemoteObject::MatrixInput_Gain][roa] = dbOcaObjectDef_MatrixInput_Gain(ch);
561 m_ROIsToDefsMap[RemoteObject::MatrixInput_Delay][roa] = dbOcaObjectDef_MatrixInput_Delay(ch);
562 m_ROIsToDefsMap[RemoteObject::MatrixInput_DelayEnable][roa] = dbOcaObjectDef_MatrixInput_DelayEnable(ch);
563 m_ROIsToDefsMap[RemoteObject::MatrixInput_EqEnable][roa] = dbOcaObjectDef_MatrixInput_EqEnable(ch);
564 m_ROIsToDefsMap[RemoteObject::MatrixInput_Polarity][roa] = dbOcaObjectDef_MatrixInput_Polarity(ch);
565 m_ROIsToDefsMap[RemoteObject::MatrixInput_ChannelName][roa] = dbOcaObjectDef_MatrixInput_ChannelName(ch);
566 m_ROIsToDefsMap[RemoteObject::MatrixInput_LevelMeterPreMute][roa] = dbOcaObjectDef_MatrixInput_LevelMeterPreMute(ch);
567 m_ROIsToDefsMap[RemoteObject::MatrixInput_LevelMeterPostMute][roa] = dbOcaObjectDef_MatrixInput_LevelMeterPostMute(ch);
568 m_ROIsToDefsMap[RemoteObject::MatrixInput_ReverbSendGain][roa] = dbOcaObjectDef_MatrixInput_ReverbSendGain(ch);
569
570 // Per-channel × mapping-area
571 for (std::uint16_t area = static_cast<std::uint16_t>(MappingAreaId::First);
572 area <= static_cast<std::uint16_t>(MappingAreaId::Fourth); ++area)
573 {
574 roa.sec = static_cast<std::int16_t>(area);
575 m_ROIsToDefsMap[RemoteObject::CoordinateMapping_SourcePosition][roa]
576 = dbOcaObjectDef_CoordinateMapping_Source_Position(area, ch);
577 }
578
579 // Per-channel × function-group
580 const auto numGroups = std::min(sc_MAX_FUNCTION_GROUPS, m_activeOutputChannelCount);
581 for (std::uint16_t fg = 1; fg <= numGroups; ++fg)
582 {
583 roa.sec = static_cast<std::int16_t>(fg);
584 m_ROIsToDefsMap[RemoteObject::SoundObjectRouting_Mute][roa]
585 = dbOcaObjectDef_SoundObjectRouting_Mute(fg, ch);
586 m_ROIsToDefsMap[RemoteObject::SoundObjectRouting_Gain][roa]
587 = dbOcaObjectDef_SoundObjectRouting_Gain(fg, ch);
588 }
589
590 // Per-channel × output-channel (matrix nodes)
591 for (std::uint16_t out = 1; out <= m_activeOutputChannelCount; ++out)
592 {
593 roa.sec = static_cast<std::int16_t>(out);
594 m_ROIsToDefsMap[RemoteObject::MatrixNode_Enable][roa] = dbOcaObjectDef_MatrixNode_Enable(ch, out);
