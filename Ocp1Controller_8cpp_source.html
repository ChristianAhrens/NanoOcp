64void Ocp1Controller::connect(const std::string& host, int port, int timeoutMs)
65{
66 if (m_state != State::Disconnected)
67 return;
68
69 m_host = host;
70 m_port = port;
71 m_timeoutMs = timeoutMs;
72
73 m_client = std::make_unique<NanoOcp1Client>(host, port, m_callbacksOnMessageThread);
74
75 m_client->onConnectionEstablished = [this]() {
76 afterConnected();
77 };
78
79 m_client->onConnectionLost = [this]() {
80 stopTimer();
81 clearPendingHandles();
82 // If disconnect() was called first, state is already Disconnected —
83 // don't transition back to Connecting. Otherwise the client retries
84 // automatically and we reflect that in the state.
85 if (m_state != State::Disconnected)
86 setState(State::Connecting);
87 };
88
89 m_client->onDataReceived = [this](const ByteVector& data) {
90 return processMessage(data);
91 };
92
93 setState(State::Connecting);
94 m_client->start();
95}

97void Ocp1Controller::disconnect()
98{
99 // Set state first so that any callback that checks m_state (e.g. the
100 // onConnectionLost lambda) sees Disconnected and does not re-enter Connecting.
101 setState(State::Disconnected);
102
103 if (m_client)
104 {
105 // Join the socket thread and the reconnect-retry timer thread BEFORE
106 // stopping the GetValues timer and BEFORE nulling the callbacks.
107 //
108 // In Soundscape mode the socket thread calls queryObjectValues() (and
109 // therefore NanoTimer::startTimer()) when the device GUID response
110 // arrives. If stopTimer() (which touches NanoTimer::m_thread) runs
111 // concurrently with startTimer() on the socket thread, the two race on
112 // m_thread — a std::thread object that is not protected by any mutex.
113 // Joining the socket thread first (via stop()) guarantees it is dead
114 // before stopTimer() ever looks at m_thread.
115 //
116 // Similarly, the reconnect-retry timer thread can be mid-way through
117 // calling onConnectionEstablished when the main thread nulls it.
118 // Joining that timer first (also via stop()) eliminates the race.
119 m_client->stop();
120
121 m_client->onConnectionEstablished = {};
122 m_client->onConnectionLost = {};
123 m_client->onDataReceived = {};
124 m_client.reset();
125 }
126
127 // Safe: socket thread is guaranteed dead, so no concurrent startTimer()
128 // call can race with this stopTimer().
129 stopTimer();
130 clearPendingHandles();
131}

163bool Ocp1Controller::createObjectSubscriptions()
164{
165 if (!m_client || m_state == State::Disconnected)
166 return false;
167
168 bool success = true;
169 for (const auto& tracked : m_trackedObjects)
170 {
171 std::uint32_t handle{0};
172 success = m_client->sendData(
173 Ocp1CommandResponseRequired(
174 tracked.def->AddSubscriptionCommand(), handle).GetSerializedData()
175 ) && success;
176 addPendingSubscriptionHandle(handle);
177 }
178
179 if (hasPendingSubscriptions())
180 setState(State::Subscribing);
181
182 return success;
183}

185bool Ocp1Controller::queryObjectValues()
186{
187 if (!m_client || m_state == State::Disconnected)
188 return false;
189
190 bool success = true;
191 for (const auto& tracked : m_trackedObjects)
192 success = queryObjectValue(*tracked.def) && success;
193
194 if (hasPendingGetValues())
195 {
196 setState(State::GetValues);
197 startTimer(m_timeoutMs * 20);
198 }
199 else
200 {
201 // No tracked objects (or all queries failed to send) — skip GetValues.
202 stopTimer();
203 if (!hasPendingSubscriptions())
204 setState(State::Connected);
205 }
206
207 return success;
208}

242void Ocp1Controller::retryPendingGetValues()
243{
244 // Collect stale ONos under the lock, then clear the map so re-issued
245 // queries register fresh handles.
246 std::vector<std::uint32_t> staleOnos;
247 {
248 std::lock_guard<std::mutex> lk(m_pendingMutex);
249 staleOnos.reserve(m_pendingGetValueHandles.size());
250 for (const auto& [handle, ono] : m_pendingGetValueHandles)
251 staleOnos.push_back(ono);
252 m_pendingGetValueHandles.clear();
253 }
254
255 if (staleOnos.empty())
256 {
257 stopTimer();
258 if (!hasPendingSubscriptions())
259 setState(State::Connected);
260 else
261 setState(State::Subscribing);
262 return;
263 }
264
265 for (const auto ono : staleOnos)
266 {
267 auto it = m_onoToIdx.find(ono);
268 if (it != m_onoToIdx.end())
269 queryObjectValue(*m_trackedObjects[it->second].def);
270 }
271}

274// ── Message dispatch ──────────────────────────────────────────────────────────
275
276bool Ocp1Controller::processMessage(const ByteVector& data)
277{
278 auto msg = Ocp1Message::UnmarshalOcp1Message(data);
279 if (!msg)
280 return false;
281
282 switch (msg->GetMessageType())
283 {
284 case Ocp1Message::Notification:
285 {
286 const auto* notif = static_cast<Ocp1Notification*>(msg.get());
287 const auto it = m_onoToIdx.find(notif->GetEmitterOno());
288 if (it == m_onoToIdx.end())
289 return false;
290 const auto& tracked = m_trackedObjects[it->second];
291 if (tracked.cb)
292 tracked.cb(notif->GetParameterData());
293 return true;
294 }
295
296 case Ocp1Message::Response:
297 {
298 const auto* resp = static_cast<Ocp1Response*>(msg.get());
299 const auto handle = resp->GetResponseHandle();
300
301 if (resp->GetResponseStatus() != 0)
302 {
303 // Error response — still counts as answered for state-advancement purposes.
304 const bool wasSub = popPendingSubscriptionHandle(handle);
305 const uint32_t failedOno = popPendingGetValueHandle(handle);
306 popPendingSetValueHandle(handle);
307
308 if (wasSub && !hasPendingSubscriptions())
309 {
310 setState(State::Subscribed);
311 if (hasPendingGetValues())
312 setState(State::GetValues);
313 else
314 {
315 stopTimer();
316 setState(State::Connected);
317 }
318 }
319 else if (failedOno != 0 && !hasPendingGetValues())
320 {
321 stopTimer();
322 setState(hasPendingSubscriptions() ? State::Subscribing : State::Connected);
323 }
324 return false;
325 }
326
327 // Subscription ACK
328 if (popPendingSubscriptionHandle(handle))
329 {
330 if (!hasPendingSubscriptions())
331 {
332 setState(State::Subscribed);
333 if (hasPendingGetValues())
334 setState(State::GetValues);
335 else
336 {
337 stopTimer();
338 setState(State::Connected);
339 }
340 }
341 return true;
342 }
343
344 // GetValue response
345 const std::uint32_t getValOno = popPendingGetValueHandle(handle);
346 if (getValOno != 0)
347 {
348 if (resp->GetParamCount() > 0)
349 {
350 const auto it = m_onoToIdx.find(getValOno);
351 if (it != m_onoToIdx.end())
352 {
353 const auto& tracked = m_trackedObjects[it->second];
354 if (tracked.cb)
355 tracked.cb(resp->GetParameterData());
356 }
357 else
358 {
359 // ONo was queried but not registered via trackObject().
360 // Let subclasses handle it (e.g. SoundscapeController intercepts
361 // the Fixed_GUID response here to trigger subscribe+query).
362 onUntrackedGetValueResponse(getValOno, resp->GetParameterData());
363 }
364 }
365 if (!hasPendingGetValues())
366 {
367 stopTimer();
368 setState(hasPendingSubscriptions() ? State::Subscribing : State::Connected);
369 }
370 return true;
371 }
372
373 // SetValue ACK (or any other tracked response)
374 popPendingSetValueHandle(handle);
375 return true;
376 }
377
378 case Ocp1Message::KeepAlive:
379 return true;
380
381 default:
382 return false;
383 }
384}

389void Ocp1Controller::addPendingSubscriptionHandle(std::uint32_t handle)
390{
391 std::lock_guard<std::mutex> lk(m_pendingMutex);
392 m_pendingSubscriptionHandles.push_back(handle);
393}
394
395bool Ocp1Controller::popPendingSubscriptionHandle(std::uint32_t handle)
396{
397 std::lock_guard<std::mutex> lk(m_pendingMutex);
398 auto it = std::find(m_pendingSubscriptionHandles.begin(),
399 m_pendingSubscriptionHandles.end(), handle);
400 if (it == m_pendingSubscriptionHandles.end())
401 return false;
402 m_pendingSubscriptionHandles.erase(it);
403 return true;
404}

406bool Ocp1Controller::hasPendingSubscriptions()
407{
