175static Demo::Config parseArgs(int argc, char** argv)
176{
177 Demo::Config cfg;
178 for (int i = 1; i < argc; ++i)
179 {
180 std::string arg = argv[i];
181
182 if (arg == "--amp")
183 {
184 cfg.mode = DemoMode::Amp;
185 }
186 else if (arg == "--soundscape" && i + 1 < argc)
187 {
188 // Preserve an already-parsed --param (order-independent parsing);
189 // otherwise this is plain overview mode.
190 cfg.mode = (cfg.mode == DemoMode::SoundscapeFocus) ? DemoMode::SoundscapeFocus
191 : DemoMode::Soundscape;
192 try { cfg.soundObject = std::stoi(argv[++i]); }
193 catch (...) { cfg.soundObject = 1; }
194 }
195 else if (arg == "--param" && i + 1 < argc)
196 {
197 cfg.mode = DemoMode::SoundscapeFocus;
198 cfg.focusParamName = argv[++i];
199 if (const auto* entry = findFocusParam(cfg.focusParamName))
200 cfg.focusParamId = entry->id;
201 }
202 else if (arg == "--addr2" && i + 1 < argc)
203 {
204 try { cfg.addr2 = std::stoi(argv[++i]); }
205 catch (...) { cfg.addr2 = 0; }
206 }
207 else if (arg == "--type" && i + 1 < argc)
208 {
209 ++i;
210 std::string t = argv[i];
211 if (t == "dx") cfg.ampType = NanoOcp1::AmpController::AmpType::Dx;
212 else if (t == "dy") cfg.ampType = NanoOcp1::AmpController::AmpType::Dy;
213 else if (t == "5d") cfg.ampType = NanoOcp1::AmpController::AmpType::FiveD;
214 }
215 else if (arg == "--ch" && i + 1 < argc)
216 {
217 try { cfg.channelCount = static_cast<std::uint16_t>(std::stoi(argv[++i])); }
218 catch (...) { cfg.channelCount = 4; }
219 }
220 else if (arg[0] != '-')
221 {
222 // positional: first is host, second is port
223 if (cfg.host == "127.0.0.1" || cfg.host.empty())
224 {
225 // treat as host unless it's a pure number
226 bool isNum = !arg.empty() && std::all_of(arg.begin(), arg.end(), ::isdigit);
227 if (!isNum)
228 cfg.host = arg;
229 else
230 {
231 try { cfg.port = std::stoi(arg); }
232 catch (...) {}
233 }
234 }
235 else
236 {
237 try { cfg.port = std::stoi(arg); }
238 catch (...) {}
239 }
240 }
241 }
242 return cfg;
243}

247int main(int argc, char** argv)
248{
249 for (int i = 1; i < argc; ++i)
250 {
251 std::string a = argv[i];
252 if (a == "-h" || a == "--help")
253 {
254 printHelp(argv[0]);
255 return 0;
256 }
257 if (a == "-v" || a == "--version")
258 {
259 std::cout << "NanoOcp1Demo " << NANOOCP1_VERSION_STRING
260 << " (NanoOcp1 " << NanoOcp1::GetVersionString() << ")\n";
261 return 0;
262 }
263 if (a == "--list-params")
264 {
265 // --list-params is exclusive to Soundscape mode (it enumerates
266 // SoundscapeController parameters), so it must be written as
267 // "--soundscape --list-params".
268 if (i == 0 || std::string(argv[i - 1]) != "--soundscape")
269 {
270 std::cerr << "--list-params must directly follow --soundscape, e.g.\n"
271 " " << argv[0] << " --soundscape --list-params\n";
272 return 1;
273 }
274 printParamList();
275 return 0;
276 }
277 }
278
279 setupTerminal();
280
281 Demo::Config cfg = parseArgs(argc, argv);
282
283 if (cfg.mode == DemoMode::SoundscapeFocus && cfg.focusParamId == SORemObjIdent::Invalid)
284 {
285 std::cerr << "Unknown --param name"
286 << (cfg.focusParamName.empty() ? "" : (": '" + cfg.focusParamName + "'"))
287 << ". Run '" << argv[0] << " --soundscape --list-params' to see valid names.\n";
288 return 1;
289 }
290
291 // Reserve the on-screen panel canvas. Amp / Soundscape-overview modes use a
292 // constant size; Soundscape-focus mode sizes itself to the current terminal
293 // height up front (redrawLoop() keeps it in sync as the terminal is resized).
294 int initialPanelLines = kPanelLines;
295 if (cfg.mode == DemoMode::SoundscapeFocus)
296 {
297 int rows = 24, cols = 80;
298 getTerminalSize(rows, cols); // best-effort; keeps the 24x80 fallback on failure
299 const int logLines = std::max(kFocusMinLogLines, rows - kFocusFixedLines - 1);
300 initialPanelLines = kFocusFixedLines + logLines;
301 g_logCapacity = logLines;
302 g_panelLines = initialPanelLines;
303 }
304 resetCanvas(initialPanelLines);
305
306 Demo demo(cfg);
307 const bool isFocusMode = (cfg.mode == DemoMode::SoundscapeFocus);
308
309 std::thread redrawThread(redrawLoop);
310
311 demo.connect();
312
313 // ── Command loop ──────────────────────────────────────────────────────────
314 std::string line;
315 while (std::getline(std::cin, line))
316 {
317 // trim whitespace
318 auto first = line.find_first_not_of(" \t");
319 const std::string trimmed = (first == std::string::npos) ? std::string()
320 : line.substr(first, line.find_last_not_of(" \t") - first + 1);
321
322 // In Soundscape-focus mode, a pending 'v' command turns the *next* line
323 // into either the new value to send, or a cancel ('b' / 'back').
324 if (isFocusMode && g_focusAwaitingInput.load())
325 {
326 g_focusAwaitingInput = false;
327 if (trimmed.empty() || trimmed == "b" || trimmed == "back")
328 pushFocusLog("Cancelled.");
329 else
330 demo.cmdSetFocusValue(trimmed);
331 printPrompt();
332 continue;
333 }
334
335 if (trimmed.empty())
336 {
337 printPrompt();
338 continue;
339 }
340
341 if (trimmed == "q" || trimmed == "quit")
342 break;
343
344 std::istringstream iss(trimmed);
345 std::string cmd;
346 iss >> cmd;
347
348 if (cmd == "c") demo.connect();
349 else if (cmd == "d") demo.disconnect();
350 else if (cmd == "a")
351 {
352 std::string addr;
353 if (iss >> addr) demo.setAddress(addr);
354 else pushLog("Usage: a <ip>");
355 }
356 else if (cmd == "p")
357 {
358 int p;
359 if (iss >> p) demo.setPort(p);
360 else pushLog("Usage: p <port>");
361 }
362 else if (cmd == "1") demo.cmdPower(true);
363 else if (cmd == "0") demo.cmdPower(false);
364 else if (cmd == "g")
365 {
366 std::uint16_t ch; float dB;
367 if (iss >> ch >> dB) demo.cmdGain(ch, dB);
368 else pushLog("Usage: g <ch> <dB> (ch: 1-4, dB: -57.5 to +6.0)");
369 }
370 else if (cmd == "m")
371 {
372 std::uint16_t ch; int mute;
373 if (iss >> ch >> mute) demo.cmdMute(ch, mute != 0);
374 else pushLog("Usage: m <ch> <1|0> (ch: 1-4)");
375 }
376 else if (cmd == "x")
377 {
378 float v;
379 if (iss >> v) demo.cmdPositionX(v);
380 else pushLog("Usage: x <meters>");
381 }
382 else if (cmd == "y")
383 {
384 float v;
385 if (iss >> v) demo.cmdPositionY(v);
386 else pushLog("Usage: y <meters>");
387 }
388 else if (cmd == "z")
389 {
390 float v;
391 if (iss >> v) demo.cmdPositionZ(v);
392 else pushLog("Usage: z <meters>");
393 }
394 else if (cmd == "sp")
395 {
396 float v;
397 if (iss >> v) demo.cmdSpread(v);
398 else pushLog("Usage: sp <0-1>");
399 }
400 else if (cmd == "dm")
401 {
402 int v;
403 if (iss >> v) demo.cmdDelayMode(v);
404 else pushLog("Usage: dm <0|1|2> (0=off 1=compensate 2=reflect)");
405 }
406 else if (cmd == "ig")
407 {
408 float v;
409 if (iss >> v) demo.cmdMatrixInputGain(v);
410 else pushLog("Usage: ig <dB> (-120.0 to +24.0)");
411 }
412 else if (cmd == "mm")
413 {
414 int v;
415 if (iss >> v) demo.cmdMatrixInputMute(v != 0);
416 else pushLog("Usage: mm <1|0>");
417 }
418 else if (cmd == "es")
419 {
420 float v;
421 if (iss >> v) demo.cmdEnSpace(v);
422 else pushLog("Usage: es <dB> (-120.0 to +24.0)");
423 }
424 else if (isFocusMode && cmd == "v")
425 {
426 g_focusAwaitingInput = true;
427 pushFocusLog("Enter new value (or 'b'/'back' to cancel):");
428 }
429 else
430 {
431 pushLog("Unknown command: " + trimmed);
432 }
433
434 printPrompt();
435 }
436
437 g_quit = true;
438 redrawThread.join();
439
440 std::cout << "\n";
441 return 0;
442}
