24void NanoTimer::startTimer(int intervalMs)
25{
26 // Serializes m_thread create/join/detach against concurrent startTimer()/stopTimer()
27 // calls from other threads. Not held while the worker thread's wait_until() runs, so
28 // joining it here can never deadlock against it re-locking m_mutex to wake up.
29 std::lock_guard<std::mutex> lifecycleLock(m_lifecycleMutex);
30
31 {
32 std::lock_guard<std::mutex> lk(m_mutex);
33 m_intervalMs = intervalMs;
34 m_deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(intervalMs);
35
36 if (!m_stop && m_thread.joinable())
37 {
38 // Already running: just push the deadline out, no thread churn.
39 m_cv.notify_all();
40 return;
41 }
42
43 m_stop = false;
44 }
45
46 // Not currently running: clean up any leftover thread before spinning up a fresh one.
47 if (m_thread.joinable())
48 {
49 if (m_thread.get_id() != std::this_thread::get_id())
50 m_thread.join();
51 else
52 m_thread.detach(); // reentrant restart from within our own timerCallback()
53 }
54
55 m_thread = std::thread([this]() {
56 std::unique_lock<std::mutex> lk(m_mutex);
57 while (true)
58 {
59 // Wait until deadline or until stop is signalled.
60 const auto waitedDeadline = m_deadline;
61 m_cv.wait_until(lk, waitedDeadline, [this, waitedDeadline]() { return m_stop || (m_deadline != waitedDeadline); });
62 if (m_stop)
63 break;
64 if (m_deadline != waitedDeadline)
65 continue; // restarted while waiting: loop around and wait on the new deadline
66
67 lk.unlock();
68 timerCallback();
69 lk.lock();
70
71 if (m_stop)
72 break;
73
74 // Reschedule relative to now, unless the callback itself already restarted us.
75 if (m_deadline == waitedDeadline)
76 m_deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(m_intervalMs);
77 }
78 });
79}

81void NanoTimer::stopTimer()
82{
83 std::lock_guard<std::mutex> lifecycleLock(m_lifecycleMutex);
84
85 {
86 std::lock_guard<std::mutex> lk(m_mutex);
87 m_stop = true;
88 }
89 m_cv.notify_all();
90
91 // Only join from outside the timer thread.
92 // If stopTimer() is called from within timerCallback(), the thread will
93 // exit its loop naturally after the callback returns and can be joined
94 // by the destructor or the next startTimer()/stopTimer() call.
95 if (m_thread.joinable() && m_thread.get_id() != std::this_thread::get_id())
96 m_thread.join();
97}
