61 void startThread(ThreadPriority priority = ThreadPriority::normal)
62 {
63 m_shouldExit.store(false);
64 m_running.store(true);
65 m_thread = std::thread([this, priority]() {
66 applyPriority(priority);
67 run();
68 m_running.store(false);
69 });
70 }

72 void stopThread(int timeoutMs)
73 {
74 m_shouldExit.store(true);
75 if (!m_thread.joinable())
76 return;
77
78 // Poll until thread exits or timeout expires, then join unconditionally.
79 auto deadline = std::chrono::steady_clock::now()
80 + std::chrono::milliseconds(timeoutMs);
81 while (m_running.load() && std::chrono::steady_clock::now() < deadline)
82 std::this_thread::sleep_for(std::chrono::milliseconds(5));
83
84 m_thread.join();
85 }
86
87 bool threadShouldExit() const noexcept { return m_shouldExit.load(); }
88 void signalThreadShouldExit() { m_shouldExit.store(true); }

102 static void applyPriority(ThreadPriority priority)
103 {
104#if defined(_WIN32) || defined(_WIN64)
105 int wp = THREAD_PRIORITY_NORMAL;
106 switch (priority)
107 {
108 case ThreadPriority::low: wp = THREAD_PRIORITY_BELOW_NORMAL; break;
109 case ThreadPriority::high: wp = THREAD_PRIORITY_ABOVE_NORMAL; break;
110 case ThreadPriority::realtime: wp = THREAD_PRIORITY_HIGHEST; break;
111 default: break;
112 }
113 SetThreadPriority(GetCurrentThread(), wp);
114#elif defined(_POSIX_THREAD_PRIORITY_SCHEDULING) && _POSIX_THREAD_PRIORITY_SCHEDULING > 0
115 int policy = SCHED_OTHER;
116 sched_param param{};
117 switch (priority)
118 {
119 case ThreadPriority::realtime:
120 policy = SCHED_FIFO;
121 param.sched_priority = sched_get_priority_max(SCHED_FIFO);
122 break;
123 case ThreadPriority::high:
124 param.sched_priority = sched_get_priority_max(SCHED_OTHER) / 2;
125 break;
126 default:
127 break;
128 }
129 pthread_setschedparam(pthread_self(), policy, &param);
130#else
131 (void)priority;
132#endif
133 }
