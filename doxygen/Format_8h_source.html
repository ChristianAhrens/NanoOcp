36static std::string fillBar(float val, float lo, float hi, int width = 16)
37{
38 const float clamped = std::max(lo, std::min(hi, val));
39 const float frac = (hi > lo) ? (clamped - lo) / (hi - lo) : 0.0f;
40 const int n = static_cast<int>(std::round(frac * static_cast<float>(width)));
41 std::string bar = "[";
42 for (int i = 0; i < width; ++i)
43 bar += (i < n) ? "\xe2\x96\x88" : "\xe2\x96\x91"; // U+2588 full / U+2591 light
44 bar += "]";
45 return bar;
46}

48// Colour-segmented VU-meter-style bar (green/yellow/red zones), redrawn on every
49// live value update from the device — used for signal-level and channel-gain
50// readouts, as opposed to fillBar()'s plain single-colour rendering for
51// positional/target controls (spread, position, delay mode).
52static std::string meterBar(float val, float lo, float hi, int width = 16)
53{
54 const float clamped = std::max(lo, std::min(hi, val));
55 const float frac = (hi > lo) ? (clamped - lo) / (hi - lo) : 0.0f;
56 const int n = static_cast<int>(std::round(frac * static_cast<float>(width)));
57 std::string bar = "[";
58 for (int i = 0; i < width; ++i)
59 {
60 if (i >= n)
61 {
62 bar += "\xe2\x96\x91"; // U+2591 light — unlit cell
63 continue;
64 }
65 const float cellFrac = static_cast<float>(i + 1) / static_cast<float>(width);
66 const char* color = (cellFrac > 0.9f) ? Ansi::Red
67 : (cellFrac > 0.75f) ? Ansi::Yellow
68 : Ansi::Green;
69 bar += std::string(color) + "\xe2\x96\x88" + Ansi::Reset; // U+2588 full, zone-coloured
70 }
71 bar += "]";
72 return bar;
73}

75// Vertical, multi-*row* counterpart to meterBar() — used by the --speakerlvls
76// loudspeaker block, where each channel is a narrow column rather than a wide
77// horizontal row, so the bar has to grow in height instead of width. Returns
78// `segments` single-character, already zone-coloured (green/yellow/red, same
79// thresholds as meterBar()) cell strings ordered top-to-bottom (index 0 is the
80// topmost/loudest segment) for one column; the caller places rows[i] into row
81// i of the grid for every column so the columns' bars line up into one
82// multi-row meter. Unknown values are returned as dim "?" placeholders.
83static std::vector<std::string> verticalBarGlyphs(bool known, float val, float lo, float hi, int segments)
84{
85 std::vector<std::string> rows(static_cast<std::size_t>(segments));
86 if (!known)
87 {
88 for (auto& r : rows) r = std::string(Ansi::Dim) + "?" + Ansi::Reset;
89 return rows;
90 }
91 const float clamped = std::max(lo, std::min(hi, val));
92 const float frac = (hi > lo) ? (clamped - lo) / (hi - lo) : 0.0f;
93 for (int i = 0; i < segments; ++i)
94 {
95 // Row i (from the top) represents the segment spanning
96 // (segments-i-1)/segments .. (segments-i)/segments of the full range.
97 const float rowTopFrac = static_cast<float>(segments - i) / static_cast<float>(segments);
98 const float rowBottomFrac = static_cast<float>(segments - i - 1) / static_cast<float>(segments);
99 if (frac <= rowBottomFrac)
100 {
101 rows[static_cast<std::size_t>(i)] = "\xe2\x96\x91"; // U+2591 light — unlit
102 continue;
103 }
104 const char* color = (rowTopFrac > 0.9f) ? Ansi::Red
105 : (rowTopFrac > 0.75f) ? Ansi::Yellow
106 : Ansi::Green;
107 rows[static_cast<std::size_t>(i)] = std::string(color) + "\xe2\x96\x88" + Ansi::Reset; // U+2588 full
108 }
109 return rows;
110}
