94bool NanoOcp1Client::start()
95{
96 m_running = true;
97
98 if (connectToSocket(getAddress(), getPort(), 50))
99 return true; // connection immediately established
100 else
101 startTimer(500); // start trying to establish connection
102
103 return false;
104}

106bool NanoOcp1Client::stop()
107{
108 m_running = false;
109
110 stopTimer();
111
112 disconnect(1000);
113
114 if (onConnectionLost && !isConnected())
115 onConnectionLost();
116
117 return !isConnected();
118}

155void NanoOcp1Client::timerCallback()
156{
157 // If already connected, short-circuit so connectToSocket is not called. The live connection is safe.
158 // If not connected, tries to connect. On success stopTimer() runs. On failure the timer keeps retrying.
159 if (!isConnected() && connectToSocket(getAddress(), getPort(), 50))
160 stopTimer();
161}

195bool NanoOcp1Server::stop()

201 }
202 else

204}

214Ocp1Connection* NanoOcp1Server::createConnectionObject()
215{
216 m_activeConnection = std::make_unique<NanoOcp1Client>(
217 m_scheduler, m_callbacksOnMessageThread, m_threadPriority);
218 m_activeConnection->onDataReceived = this->onDataReceived;
219
220 return m_activeConnection.get();
