44 struct Config
45 {
46 std::string host{"127.0.0.1"};
47 int port{50014};
48 DemoMode mode{DemoMode::Amp};
49 // Amp-specific
50 NanoOcp1::AmpController::AmpType ampType{NanoOcp1::AmpController::AmpType::Dy};

52 // DS100-specific: sound-object range for Soundscape-overview mode
53 // (soundObjectHi == soundObjectLo means a single object).

55 int soundObjectHi{1};
56 // Soundscape-focus-specific (only used when mode == SoundscapeFocus);

61 // Soundscape-only: show the optional loudspeaker level-meter block

64 bool speakerLvls{false};

67 explicit Demo(Config cfg) : m_cfg(std::move(cfg))
68 {
69 std::lock_guard<std::mutex> lk(g_stateMutex);
70 g_state.address = m_cfg.host;
71 g_state.port = m_cfg.port;
72 g_state.mode = m_cfg.mode;
73 g_state.amp.type = m_cfg.ampType;
74 g_state.amp.channelCount = static_cast<int>(m_cfg.channelCount);
75 g_state.ds100.soundObjectLo = m_cfg.soundObjectLo;
76 g_state.ds100.soundObjectHi = m_cfg.soundObjectHi;
77 g_state.ds100.view = SoundscapeView::Overview;
78 g_state.ds100.objects.clear();
79 for (int so = m_cfg.soundObjectLo; so <= m_cfg.soundObjectHi; ++so)
80 {
81 AppState::SoObjState o;
82 o.soundObject = so;
83 g_state.ds100.objects.push_back(o);
84 }
85 g_state.ds100.speakerLvlsEnabled = m_cfg.speakerLvls;
86 g_state.ds100.speakerObjects.clear();
87 if (m_cfg.speakerLvls)
88 {
89 const int maxOut = static_cast<int>(NanoOcp1::SoundscapeController::sc_MAX_OUTPUT_CHANNELS);
90 for (int ch = 1; ch <= maxOut; ++ch)
91 {
92 AppState::SpeakerObjState s;
93 s.outputChannel = ch;
94 g_state.ds100.speakerObjects.push_back(s);
95 }
96 }
97 g_state.focus.paramName = m_cfg.focusParamName;
98 g_state.focus.paramId = m_cfg.focusParamId;
99 g_state.focus.addr = SORemObjAddr{static_cast<std::int16_t>(m_cfg.soundObjectLo),
100 static_cast<std::int16_t>(m_cfg.addr2)};
101 }

107 void connect()
108 {
109 teardown();
110
111 switch (m_cfg.mode)
112 {
113 case DemoMode::Amp:
114 connectAmp();
115 break;
116 case DemoMode::Soundscape:
117 if (m_routingActive) connectRouting();
118 else connectDS100();
119 break;
120 case DemoMode::SoundscapeFocus:
121 connectFocus();
122 break;
123 }
124 }
125
126 void disconnect()
127 {
128 teardown();
129 {
130 std::lock_guard<std::mutex> lk(g_stateMutex);
131 g_state.ctrlState = CtrlState::Disconnected;
132 switch (m_cfg.mode)
133 {
134 case DemoMode::Amp:
135 resetAmpDisplay();
136 break;
137 case DemoMode::Soundscape:
138 if (m_routingActive) resetRoutingDisplay();
139 else resetSoDisplay();
140 resetSpeakerDisplay();
141 break;
142 case DemoMode::SoundscapeFocus:
143 resetFocusDisplay();
144 break;
145 }
146 }
147 g_needsRedraw = true;
148 if (m_cfg.mode == DemoMode::SoundscapeFocus)
149 pushFocusLog("Disconnected.");
150 else
151 pushLog("Disconnected.");
152 }

307 // ── Soundobject-Routing sub-mode ──────────────────────────────────────────
308 // Entered (for any sound-object range addressable via --soundscape, single
309 // or a sub-range of it) to control SoundObjectRouting_Gain/Mute for every
310 // selected sound object across all 32 Function Groups. Switching into or
311 // out of this sub-mode disconnects and re-subscribes, since
312 // SoundscapeController's active remote-object list can only be changed
313 // while disconnected.
314
315 void enterRouting(int lo, int hi)
316 {
317 if (m_cfg.mode != DemoMode::Soundscape)
318 {
319 pushLog("Routing sub-mode is only available in Soundscape mode.");
320 return;
321 }
322 if (lo > hi) std::swap(lo, hi);
323 if (lo < m_cfg.soundObjectLo || hi > m_cfg.soundObjectHi)
324 {
325 pushLog("Sound object range out of bounds of the active range ("
326 + std::to_string(m_cfg.soundObjectLo) + "-" + std::to_string(m_cfg.soundObjectHi) + ")");
327 return;
328 }
329 teardown();
330 m_routingActive = true;
331 m_routingLo = lo;
332 m_routingHi = hi;
333 connectRouting();
334 }
335
336 void exitRouting()
337 {
338 if (!m_routingActive) { pushLog("Not in routing sub-mode."); return; }
339 teardown();
340 m_routingActive = false;
341 connectDS100();
342 }

387 // ── Soundscape-focus command ──────────────────────────────────────────────
388
389 // Parses `rawValue` according to the focused parameter's last-known data type
390 // and sends it as a SetValue. Requires at least one value to have already
391 // been received, since that is how the demo learns the parameter's type.
392 void cmdSetFocusValue(const std::string& rawValue)
393 {
394 if (!m_ds100) { pushFocusLog("Not connected."); return; }
395
396 NanoOcp1::Ocp1DataType hint = NanoOcp1::OCP1DATATYPE_NONE;
397 SORemObjAddr addr;
398 {
399 std::lock_guard<std::mutex> lk(g_stateMutex);
400 if (g_state.focus.valueKnown)
401 hint = g_state.focus.lastValue.GetDataType();
402 addr = g_state.focus.addr;
403 }
404
405 NanoOcp1::Variant value;
406 std::string err;
407 if (!parseVariantInput(rawValue, hint, value, err))
408 {
409 pushFocusLog("Invalid value: " + err);
410 return;
411 }
412
413 using RO = NanoOcp1::SoundscapeController::RemoteObject;
414 if (!m_ds100->setObjectValue(RO{m_cfg.focusParamId, addr, value}))
415 pushFocusLog("setObjectValue(" + m_cfg.focusParamName + ") failed (not connected?)");
416 else
417 pushFocusLog("Sent " + m_cfg.focusParamName + " -> " + formatVariant(value));
418 }

525 // ── DS100 connection (Soundscape-overview, ranged) ────────────────────────
526
527 void connectDS100()
528 {
529 using ROI = NanoOcp1::SoundscapeController::RemoteObject::RemObjIdent;
530 using ROA = NanoOcp1::SoundscapeController::RemObjAddr;
531 using RO = NanoOcp1::SoundscapeController::RemoteObject;
532
533 m_ds100 = std::make_unique<NanoOcp1::SoundscapeController>(m_scheduler);
534
535 std::vector<RO> active;
536 for (int so = m_cfg.soundObjectLo; so <= m_cfg.soundObjectHi; ++so)
537 {
538 const auto a = static_cast<std::int16_t>(so);
539 active.push_back(RO{ ROI::MatrixInput_LevelMeterIn, ROA{a, 0} });
540 active.push_back(RO{ ROI::MatrixInput_Gain, ROA{a, 0} });
541 active.push_back(RO{ ROI::MatrixInput_Mute, ROA{a, 0} });
542 active.push_back(RO{ ROI::Positioning_SourcePosition, ROA{a, 0} });
543 active.push_back(RO{ ROI::Positioning_SourceSpread, ROA{a, 0} });
544 active.push_back(RO{ ROI::Positioning_SourceDelayMode, ROA{a, 0} });
545 active.push_back(RO{ ROI::MatrixInput_ReverbSendGain, ROA{a, 0} });
546 }
547 addSpeakerLvlsSubscriptions(active);
548 m_ds100->setActiveRemoteObjects(active);
549
550 {
551 std::lock_guard<std::mutex> lk(g_stateMutex);
552 g_state.ds100.view = SoundscapeView::Overview;
553 }
554 g_needsRedraw = true;
555
556 m_ds100->onStateChanged = [this](CtrlState s) {
557 std::string devModel;
558 if (s == CtrlState::Connected)
559 {
560 devModel = modelStr(m_ds100->getConnectedDeviceModel());
561 const int stack = m_ds100->getOcaStackIdent();
562 if (stack >= 0)
563 devModel += " stack " + std::to_string(stack);
564 }
565 {
566 std::lock_guard<std::mutex> lk(g_stateMutex);
567 g_state.ctrlState = s;
568 g_state.ds100.deviceModel = devModel;
569 if (s == CtrlState::Disconnected)
570 {
571 resetSoDisplay();
572 resetSpeakerDisplay();
573 }
574 }
575 g_needsRedraw = true;
576 pushLog("State: " + stateToStr(s));
577 };
578
579 m_ds100->onRemoteObjectReceived = [this](const NanoOcp1::SoundscapeController::RemoteObject& ro) -> bool {
580 using Id = NanoOcp1::SoundscapeController::RemoteObject::RemObjIdent;
581 bool ok = false;
582 bool logPosition = false;
583 std::ostringstream logMsg;
584 {
585 std::lock_guard<std::mutex> lk(g_stateMutex);
586 auto& ds = g_state.ds100;
587
588 // Speaker-level-meter objects are addressed by output channel,
589 // not sound object, so they're handled up front — before the
590 // sound-object range bounds check below, which would otherwise
591 // incorrectly reject any output channel outside that range.
592 if (m_cfg.speakerLvls
593 && (ro.Id == Id::Positioning_SpeakerPosition || ro.Id == Id::MatrixOutput_LevelMeterPostMute))
594 {
595 ok = updateSpeakerObject(ro);
596 if (ok) g_needsRedraw = true;
597 return ok;
598 }
599
600 const int idx = ro.Addr.pri - ds.soundObjectLo;
601 if (idx < 0 || idx >= static_cast<int>(ds.objects.size()))
602 return false;
603 auto& o = ds.objects[idx];
