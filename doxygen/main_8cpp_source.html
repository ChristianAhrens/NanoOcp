258// Parses a --soundscape argument of either "<N>" or "<lo>-<hi>" form. Returns
259// false (leaving lo/hi unchanged) if the token isn't a valid integer or range.
260static bool parseSoundObjectRange(const std::string& tok, int& lo, int& hi)
261{
262 const auto dash = tok.find('-');
263 if (dash == std::string::npos || dash == 0)
264 {
265 try { lo = hi = std::stoi(tok); }
266 catch (...) { return false; }
267 return true;
268 }
269 try
270 {
271 lo = std::stoi(tok.substr(0, dash));
272 hi = std::stoi(tok.substr(dash + 1));
273 }
274 catch (...) { return false; }
275 return true;
276}

278static Demo::Config parseArgs(int argc, char** argv)
279{
280 Demo::Config cfg;
281 for (int i = 1; i < argc; ++i)
282 {
283 std::string arg = argv[i];
284
285 if (arg == "--amp")
286 {
287 cfg.mode = DemoMode::Amp;
288 }
289 else if (arg == "--soundscape" && i + 1 < argc)
290 {
291 // Preserve an already-parsed --param (order-independent parsing);
292 // otherwise this is plain overview mode.
293 cfg.mode = (cfg.mode == DemoMode::SoundscapeFocus) ? DemoMode::SoundscapeFocus
294 : DemoMode::Soundscape;
295 int lo = 1, hi = 1;
296 if (parseSoundObjectRange(argv[++i], lo, hi))
297 {
298 cfg.soundObjectLo = lo;
299 cfg.soundObjectHi = hi;
300 }
301 else
302 {
303 cfg.soundObjectLo = cfg.soundObjectHi = 1;
304 }
305 }
306 else if (arg == "--param" && i + 1 < argc)
307 {
308 cfg.mode = DemoMode::SoundscapeFocus;
309 cfg.focusParamName = argv[++i];
310 if (const auto* entry = findFocusParam(cfg.focusParamName))
311 cfg.focusParamId = entry->id;
312 }
313 else if (arg == "--addr2" && i + 1 < argc)
314 {
315 try { cfg.addr2 = std::stoi(argv[++i]); }
316 catch (...) { cfg.addr2 = 0; }
317 }
318 else if (arg == "--type" && i + 1 < argc)
319 {
320 ++i;
321 std::string t = argv[i];
322 if (t == "dx") cfg.ampType = NanoOcp1::AmpController::AmpType::Dx;
323 else if (t == "dy") cfg.ampType = NanoOcp1::AmpController::AmpType::Dy;
324 else if (t == "5d") cfg.ampType = NanoOcp1::AmpController::AmpType::FiveD;
325 }
326 else if (arg == "--ch" && i + 1 < argc)
327 {
328 try { cfg.channelCount = static_cast<std::uint16_t>(std::stoi(argv[++i])); }
329 catch (...) { cfg.channelCount = 4; }
330 }
331 else if (arg == "--speakerlvls")
332 {
333 cfg.speakerLvls = true;
334 }
335 else if (arg[0] != '-')
336 {
337 // positional: first is host, second is port
338 if (cfg.host == "127.0.0.1" || cfg.host.empty())
339 {
340 // treat as host unless it's a pure number
341 bool isNum = !arg.empty() && std::all_of(arg.begin(), arg.end(), ::isdigit);
342 if (!isNum)
343 cfg.host = arg;
344 else
345 {
346 try { cfg.port = std::stoi(arg); }
347 catch (...) {}
348 }
349 }
350 else
351 {
352 try { cfg.port = std::stoi(arg); }
353 catch (...) {}
354 }
355 }
356 }
357 return cfg;
358}

362int main(int argc, char** argv)
363{
364 for (int i = 1; i < argc; ++i)
365 {
366 std::string a = argv[i];
367 if (a == "-h" || a == "--help")
368 {
369 printHelp(argv[0]);
370 return 0;
371 }
372 if (a == "-v" || a == "--version")
373 {
374 std::cout << "NanoOcp1Demo " << NANOOCP1_VERSION_STRING
375 << " (NanoOcp1 " << NanoOcp1::GetVersionString() << ")\n";
376 return 0;
377 }
378 if (a == "--list-params")
379 {
380 // --list-params is exclusive to Soundscape mode (it enumerates
381 // SoundscapeController parameters), so it must be written as
382 // "--soundscape --list-params".
383 if (i == 0 || std::string(argv[i - 1]) != "--soundscape")
384 {
385 std::cerr << "--list-params must directly follow --soundscape, e.g.\n"
386 " " << argv[0] << " --soundscape --list-params\n";
387 return 1;
388 }
389 printParamList();
390 return 0;
391 }
392 }
393
394 setupTerminal();
395
396 Demo::Config cfg = parseArgs(argc, argv);
397
398 if (cfg.mode == DemoMode::SoundscapeFocus && cfg.focusParamId == SORemObjIdent::Invalid)
399 {
400 std::cerr << "Unknown --param name"
401 << (cfg.focusParamName.empty() ? "" : (": '" + cfg.focusParamName + "'"))
402 << ". Run '" << argv[0] << " --soundscape --list-params' to see valid names.\n";
403 return 1;
404 }
405
406 if (cfg.mode == DemoMode::SoundscapeFocus && cfg.soundObjectHi != cfg.soundObjectLo)
407 {
408 std::cerr << "A sound-object range (<lo>-<hi>) cannot be combined with --param; "
409 "Soundscape-focus mode addresses exactly one object.\n";
410 return 1;
411 }
412
413 if ((cfg.mode == DemoMode::Soundscape || cfg.mode == DemoMode::SoundscapeFocus)
414 && (cfg.soundObjectLo < 1
415 || cfg.soundObjectHi > static_cast<int>(NanoOcp1::SoundscapeController::sc_MAX_INPUT_CHANNELS)
416 || cfg.soundObjectLo > cfg.soundObjectHi))
417 {
418 std::cerr << "Invalid --soundscape range (expected 1-"
419 << NanoOcp1::SoundscapeController::sc_MAX_INPUT_CHANNELS << ", lo <= hi).\n";
420 return 1;
421 }
422
423 // Reserve the on-screen panel canvas, sized for the current terminal.
424 // computePanelFit() also tells us up front whether this terminal is too
425 // small for the requested mode, in which case a short warning panel is
426 // reserved instead (redrawLoop() keeps all of this in sync thereafter, as
427 // the terminal is resized or the user enters/leaves the routing sub-mode).
428 int rows = 24, cols = 80;
429 getTerminalSize(rows, cols); // best-effort; keeps the 24x80 fallback on failure
430 g_termRows = rows;
431 g_termCols = cols;
432
433 const int overviewRangeCount = cfg.soundObjectHi - cfg.soundObjectLo + 1;
434 // Nothing is known about which outputs have an assigned loudspeaker
435 // position yet at startup (that arrives only after connecting), so the
436 // speaker block's column count starts from 0 and self-corrects within the
437 // first few redraw ticks once Positioning_SpeakerPosition values arrive.
438 const PanelFit fit = computePanelFit(cfg.mode, SoundscapeView::Overview, overviewRangeCount, 1,
439 cfg.speakerLvls, 0, rows, cols);
440 g_sizeFits = fit.fits;
441 g_neededRows = fit.neededRows;
442 g_neededCols = fit.neededCols;
443 if (cfg.mode == DemoMode::Soundscape)
444 {
445 g_soColsPerBand = fit.colsPerBand;
446 if (cfg.speakerLvls) g_spkColsPerBand = fit.spkColsPerBand;
447 }
448
449 int initialPanelLines;
450 if (!fit.fits)
451 {
452 initialPanelLines = kWarningPanelLines;
453 }
454 else if (cfg.mode == DemoMode::SoundscapeFocus)
455 {
456 const int logLines = std::max(kFocusMinLogLines, rows - kFocusFixedLines - 1);
457 initialPanelLines = kFocusFixedLines + logLines;
458 g_logCapacity = logLines;
459 }
460 else if (cfg.mode == DemoMode::Soundscape)
461 {
462 initialPanelLines = fit.neededRows - 1; // neededRows includes the prompt row; panelLines doesn't
463 }
464 else // Amp
465 {
466 initialPanelLines = kPanelLines;
467 }
468 g_panelLines = initialPanelLines;
469 resetCanvas(initialPanelLines);
470
471 Demo demo(cfg);
472 const bool isFocusMode = (cfg.mode == DemoMode::SoundscapeFocus);
473
474 std::thread redrawThread(redrawLoop);
475
476 demo.connect();
477
478 // ── Command loop ──────────────────────────────────────────────────────────
479 std::string line;
480 while (std::getline(std::cin, line))
481 {
482 // trim whitespace
483 auto first = line.find_first_not_of(" \t");
484 const std::string trimmed = (first == std::string::npos) ? std::string()
485 : line.substr(first, line.find_last_not_of(" \t") - first + 1);
486
487 // In Soundscape-focus mode, a pending 'v' command turns the *next* line
488 // into either the new value to send, or a cancel ('b' / 'back').
489 if (isFocusMode && g_focusAwaitingInput.load())
490 {
491 g_focusAwaitingInput = false;
492 if (trimmed.empty() || trimmed == "b" || trimmed == "back")
493 pushFocusLog("Cancelled.");
494 else
495 demo.cmdSetFocusValue(trimmed);
496 printPrompt();
497 continue;
498 }
499
500 if (trimmed.empty())
501 {
502 printPrompt();
503 continue;
504 }
505
506 if (trimmed == "q" || trimmed == "quit")
507 break;
508
509 std::istringstream iss(trimmed);
510 std::string cmd;
511 iss >> cmd;
512
513 if (cmd == "c") demo.connect();
514 else if (cmd == "d") demo.disconnect();
515 else if (cmd == "a")
516 {
517 std::string addr;
518 if (iss >> addr) demo.setAddress(addr);
519 else pushLog("Usage: a <ip>");
520 }
521 else if (cmd == "p")
522 {
523 int p;
524 if (iss >> p) demo.setPort(p);
525 else pushLog("Usage: p <port>");
526 }
527 else if (cmd == "1") demo.cmdPower(true);
528 else if (cmd == "0") demo.cmdPower(false);
529 else if (cmd == "g")
530 {
531 std::uint16_t ch; float dB;
532 if (iss >> ch >> dB) demo.cmdGain(ch, dB);
533 else pushLog("Usage: g <ch> <dB> (ch: 1-4, dB: -57.5 to +6.0)");
534 }
535 else if (cmd == "m")
536 {
537 std::uint16_t ch; int mute;
538 if (iss >> ch >> mute) demo.cmdMute(ch, mute != 0);
539 else pushLog("Usage: m <ch> <1|0> (ch: 1-4)");
540 }
541 else if (cmd == "x")
542 {
543 float a;
544 if (iss >> a) { float b; if (iss >> b) demo.cmdPositionX(b, static_cast<int>(a)); else demo.cmdPositionX(a); }
545 else pushLog("Usage: x [so] <meters>");
546 }
547 else if (cmd == "y")
548 {
549 float a;
550 if (iss >> a) { float b; if (iss >> b) demo.cmdPositionY(b, static_cast<int>(a)); else demo.cmdPositionY(a); }
551 else pushLog("Usage: y [so] <meters>");
552 }
553 else if (cmd == "z")
554 {
555 float a;
556 if (iss >> a) { float b; if (iss >> b) demo.cmdPositionZ(b, static_cast<int>(a)); else demo.cmdPositionZ(a); }
557 else pushLog("Usage: z [so] <meters>");
558 }
559 else if (cmd == "sp")
560 {
561 float a;
562 if (iss >> a) { float b; if (iss >> b) demo.cmdSpread(b, static_cast<int>(a)); else demo.cmdSpread(a); }
563 else pushLog("Usage: sp [so] <0-1>");
564 }
565 else if (cmd == "dm")
566 {
567 int a;
568 if (iss >> a) { int b; if (iss >> b) demo.cmdDelayMode(b, a); else demo.cmdDelayMode(a); }
569 else pushLog("Usage: dm [so] <0|1|2> (0=off 1=compensate 2=reflect)");
570 }
571 else if (cmd == "ig")
572 {
573 float a;
574 if (iss >> a) { float b; if (iss >> b) demo.cmdMatrixInputGain(b, static_cast<int>(a)); else demo.cmdMatrixInputGain(a); }
575 else pushLog("Usage: ig [so] <dB> (-120.0 to +24.0)");
576 }
577 else if (cmd == "mm")
578 {
579 int a;
580 if (iss >> a) { int b; if (iss >> b) demo.cmdMatrixInputMute(b != 0, a); else demo.cmdMatrixInputMute(a != 0); }
581 else pushLog("Usage: mm [so] <1|0>");
582 }
583 else if (cmd == "es")
584 {
585 float a;
586 if (iss >> a) { float b; if (iss >> b) demo.cmdEnSpace(b, static_cast<int>(a)); else demo.cmdEnSpace(a); }
587 else pushLog("Usage: es [so] <dB> (-120.0 to +24.0)");
588 }
589 else if (cmd == "r")
590 {
591 std::string tok;
592 int lo, hi;
593 if (iss >> tok && parseSoundObjectRange(tok, lo, hi)) demo.enterRouting(lo, hi);
594 else pushLog("Usage: r <so> | r <lo>-<hi>");
595 }
596 else if (cmd == "rg")
597 {
598 int fg;
599 if (iss >> fg)
600 {
601 float a;
602 if (iss >> a)
603 {
604 float b;
605 if (iss >> b) demo.cmdRoutingGain(fg, b, static_cast<int>(a));
606 else demo.cmdRoutingGain(fg, a);
607 }
608 else pushLog("Usage: rg <fg> [so] <dB> (-120.0 to +24.0)");
609 }
610 else pushLog("Usage: rg <fg> [so] <dB> (-120.0 to +24.0)");
611 }
612 else if (cmd == "rm")
613 {
614 int fg;
615 if (iss >> fg)
616 {
617 int a;
618 if (iss >> a)
619 {
620 int b;
621 if (iss >> b) demo.cmdRoutingMute(fg, b != 0, a);
622 else demo.cmdRoutingMute(fg, a != 0);
623 }
624 else pushLog("Usage: rm <fg> [so] <1|0>");
625 }
626 else pushLog("Usage: rm <fg> [so] <1|0>");
627 }
628 else if (cmd == "b" || cmd == "back")
629 {
630 demo.exitRouting();
631 }
632 else if (isFocusMode && cmd == "v")
633 {
634 g_focusAwaitingInput = true;
635 pushFocusLog("Enter new value (or 'b'/'back' to cancel):");
636 }
637 else
638 {
639 pushLog("Unknown command: " + trimmed);
640 }
641
642 printPrompt();
643 }
644
645 g_quit = true;
646 redrawThread.join();
647
648 std::cout << "\n";
649 return 0;
650}
