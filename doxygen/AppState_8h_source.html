41// ── Panel dimensions ──────────────────────────────────────────────────────────
42// Amp mode uses exactly 14 fixed rows + 5 log rows = 19 total, never resized.
43// Layout (fixed rows):
44// 1 title 2 sep 3 host/port
45// 4 status 5 mode row 1 6-9 mode rows 2-5
46// 10 sep 11-13 commands 14 Events header
47static constexpr int kLogLines = 5;
48static constexpr int kFixedLines = 14;
49static constexpr int kPanelLines = kFixedLines + kLogLines; // 19
50
51// Soundscape-overview mode lays out one channel-strip *column* per sound
52// object (parameter rows stacked top-to-bottom within each column), grouped
53// into left-to-right "bands" that wrap onto additional stacked bands when the
54// active range doesn't fit the terminal width in one band. Each band is
55// kSoBandRows tall (a header row of SO numbers, a multi-cell level-meter bar
56// row, then one row per remaining parameter); bands are separated by a thin
57// rule. See soColumnsPerBand() / soOverviewPanelLines().
58// 1 title 2 sep 3 host/port 4 status 5 sep
59// (band 1: kSoBandRows rows) [sep (band 2: kSoBandRows rows)]...
60// sep 4 command rows Events header
61static constexpr int kSoHeaderLines = 5;
62static constexpr int kSoFooterLines = 6; // sep + 4 cmd rows + Events header
63static constexpr int kSoLabelGutter = 6; // left-hand row-label column width
64static constexpr int kSoValueWidth = 9; // per-SO value field width (bar: "[" + 7 cells + "]")
65static constexpr int kSoLevelBarCells = 7; // meterBar() cell count, sized to fit kSoValueWidth
66static constexpr int kSoColumnStride = kSoValueWidth + 1; // + 1-space separator
67static constexpr int kSoBandRows = 11; // SO header + Lvl(bar) + dB + 8 more parameter rows
68
69// How many sound-object columns fit side by side in one band, given the
70// current terminal width.
71static inline int soColumnsPerBand(int termWidth)
72{
73 return std::max(1, (termWidth - kSoLabelGutter) / kSoColumnStride);
74}
75
76static inline int soOverviewPanelLines(int rangeCount, int colsPerBand)
77{
78 const int numBands = std::max(1, (std::max(1, rangeCount) + colsPerBand - 1) / colsPerBand);
79 return kSoHeaderLines + numBands * kSoBandRows + (numBands - 1) + kSoFooterLines + kLogLines;
80}

82// Soundscape-routing sub-mode lays out one column per *selected* sound object
83// (a sub-range of the active overview range) — the same column/band scheme as
84// Soundscape-overview, just with function groups as rows instead of a fixed
85// parameter list: a header row of SO numbers, then one row per function group
86// (always all 32; dimmed and collapsed when FunctionGroup_Mode says the group
87// is unused) showing that group's routing gain + mute per SO column, combined
88// into a single cell. See soRoutingColumnsPerBand() / soRoutingPanelLines().
89// 1 title 2 sep 3 host/port 4 status 5 sep
90// (band 1: kRoutingBandRows rows) [sep (band 2: kRoutingBandRows rows)]...
91// sep 2 command rows Events header
92static constexpr int kRoutingHeaderLines = 5;
93static constexpr int kRoutingFooterLines = 4; // sep + 2 cmd rows + Events header
94static constexpr int kRoutingLabelGutter = 20; // "<fg#> <name, truncated>"
95static constexpr int kRoutingValueWidth = 8; // "<gainDb:6.1f> <mute-glyph>"
96static constexpr int kRoutingColumnStride = kRoutingValueWidth + 1; // + 1-space separator
97static constexpr int kRoutingBandRows = 1 + static_cast<int>(NanoOcp1::SoundscapeController::sc_MAX_FUNCTION_GROUPS); // SO header + 32 FG rows
98
99// How many sound-object columns fit side by side in one routing band, given
100// the current terminal width.
101static inline int soRoutingColumnsPerBand(int termWidth)
102{
103 return std::max(1, (termWidth - kRoutingLabelGutter) / kRoutingColumnStride);
104}
105
106static inline int soRoutingPanelLines(int rangeCount, int colsPerBand)
107{
108 const int numBands = std::max(1, (std::max(1, rangeCount) + colsPerBand - 1) / colsPerBand);
109 return kRoutingHeaderLines + numBands * kRoutingBandRows + (numBands - 1) + kRoutingFooterLines + kLogLines;
110}

112// Optional (--speakerlvls) loudspeaker level-meter block, appended below
113// whichever Soundscape view (overview or routing) is currently shown — same
114// column/band scheme again, one column per output channel that has an
115// assigned loudspeaker position, with a short *vertical* meter bar (several
116// stacked rows, tallest segment on top) instead of a horizontal one, since
117// the block sits below a much wider panel and columns are narrow. See
118// soSpeakerColumnsPerBand() / soSpeakerBlockLines().

122static constexpr int kSpkLabelGutter = 5;
123static constexpr int kSpkValueWidth = 6;
124static constexpr int kSpkColumnStride = kSpkValueWidth + 1; // + 1-space separator
125static constexpr int kSpkBarSegments = 3; // "at least three blocks in height"
126static constexpr int kSpkBandRows = 1 + kSpkBarSegments + 1; // Out header + bar + dB

128static inline int soSpeakerMinCols() { return kSpkLabelGutter + kSpkColumnStride; }

130static inline int soSpeakerColumnsPerBand(int termWidth)

135static inline int soSpeakerBlockLines(int assignedCount, int colsPerBand)

205 int neededRows{0}; // includes the one row the input prompt itself needs
206 int neededCols{0};

208 int spkColsPerBand{1}; // meaningful only when the --speakerlvls block is shown

211// Determines whether a termRows x termCols terminal can fit the current
212// mode's panel — plus, when `speakerLvlsEnabled`, the optional loudspeaker
213// level-meter block appended below it (Soundscape overview/routing only) —
214// and how many columns per band each of those implies. Used both at startup
215// (main.cpp) and on every redraw tick (Panels.h's redrawLoop()) so a live
216// terminal resize, or the speaker block's column count changing as
217// Positioning_SpeakerPosition values arrive, is caught the same way a
218// too-small starting size is.
219static inline PanelFit computePanelFit(DemoMode mode, SoundscapeView view,
220 int overviewRangeCount, int routingRangeCount,
221 bool speakerLvlsEnabled, int assignedSpeakerCount,
222 int termRows, int termCols)
223{
224 PanelFit f;
225 int mainRows = 0, mainCols = 0;
226 switch (mode)
227 {
228 case DemoMode::Amp:
229 mainCols = kMinTextCols;
230 mainRows = kPanelLines;
231 break;
232
233 case DemoMode::SoundscapeFocus:
234 mainCols = kMinTextCols;
235 mainRows = kFocusFixedLines + kFocusMinLogLines;
236 break;
237
238 case DemoMode::Soundscape:
239 if (view == SoundscapeView::Routing)
240 {
241 mainCols = soRoutingMinCols();
242 f.colsPerBand = (termCols >= mainCols) ? soRoutingColumnsPerBand(termCols) : 1;
243 mainRows = soRoutingPanelLines(routingRangeCount, f.colsPerBand);
244 }
245 else
246 {
247 mainCols = soOverviewMinCols();
248 f.colsPerBand = (termCols >= mainCols) ? soColumnsPerBand(termCols) : 1;
249 mainRows = soOverviewPanelLines(overviewRangeCount, f.colsPerBand);
250 }
251 break;
252 }
253
254 int spkRows = 0, spkCols = 0;
255 if (mode == DemoMode::Soundscape && speakerLvlsEnabled)
256 {
257 spkCols = soSpeakerMinCols();
258 f.spkColsPerBand = (termCols >= spkCols) ? soSpeakerColumnsPerBand(termCols) : 1;
259 spkRows = soSpeakerBlockLines(assignedSpeakerCount, f.spkColsPerBand);
260 }
261
262 f.neededCols = std::max(mainCols, spkCols);
263 f.neededRows = mainRows + spkRows + 1; // +1 for the input prompt row
264 f.fits = (termRows >= f.neededRows) && (termCols >= f.neededCols);
265 return f;
266}

409// Returns "HH:MM:SS.mmm" for the current local time — used to timestamp entries
410// in Soundscape-focus mode's monitor log.
411static std::string timestampNow()
412{
413 using namespace std::chrono;
414 const auto now = system_clock::now();
415 const auto t = system_clock::to_time_t(now);
416 const auto ms = duration_cast<milliseconds>(now.time_since_epoch()) % 1000;
417 std::tm tmBuf{};
418#if defined(_WIN32) || defined(_WIN64)
419 localtime_s(&tmBuf, &t);
420#else
421 localtime_r(&t, &tmBuf);
422#endif
423 std::ostringstream oss;
424 oss << std::put_time(&tmBuf, "%H:%M:%S") << '.' << std::setw(3) << std::setfill('0') << ms.count();
425 return oss.str();
426}
