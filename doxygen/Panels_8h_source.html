41// Each panel renderer emits exactly kFixedLines (or, for Soundscape-focus,
42// kFocusFixedLines + logCapacity) newline-terminated rows. Must be called with
43// g_ioMutex held.
44
45static std::string makeSep()
46{
47 std::string s;
48 s.reserve(50 * 3);
49 for (int i = 0; i < 50; ++i)
50 s += "\xe2\x94\x80"; // U+2500 BOX DRAWINGS LIGHT HORIZONTAL
51 return s;
52}

54static void renderStatusRow(std::ostringstream& o, CtrlState cs)
55{
56 o << " Status ";
57 if (cs == CtrlState::Connected)
58 o << Ansi::Green << "\xe2\x97\x8f " << Ansi::Reset; // U+25CF filled circle
59 else if (cs == CtrlState::Disconnected)
60 o << Ansi::Red << "\xe2\x97\x8b " << Ansi::Reset; // U+25CB open circle
61 else
62 o << Ansi::Yellow << "\xe2\x97\x8b " << Ansi::Reset;
63 o << stateToStr(cs);
64}

66// Renders the "Events" header + a fixed-capacity scrolling log — always the
67// very last section of a panel, immediately above the input prompt. Shared by
68// every mode's renderer (Amp, Soundscape overview/routing — after any
69// optional trailing block such as --speakerlvls's — and Soundscape-focus,
70// which passes its own dynamically-sized logCapacity instead of kLogLines).
71// Emits exactly 1 + capacity rows.
72static void renderEventsSection(const std::vector<std::string>& log, int capacity)
73{
74 auto row = [](const std::string& s) {
75 std::cout << s << Ansi::Eol << "\n";
76 };
77 row(std::string(Ansi::Dim) + " Events" + Ansi::Reset);
78 for (int i = 0; i < capacity; ++i)
79 {
80 if (i < static_cast<int>(log.size()))
81 row(std::string(" ") + Ansi::Cyan + log[i] + Ansi::Reset);
82 else
83 row("");
84 }
85}

87// Renders a short, centred warning in place of the normal panel when the
88// current terminal is too small for the requested cmdln interface (see
89// computePanelFit() in AppState.h). Emits exactly kWarningPanelLines rows.
90static void renderTooSmallWarning(DemoMode mode, SoundscapeView view,
91 int neededRows, int neededCols,
92 int haveRows, int haveCols)
93{
94 auto row = [](const std::string& s) {
95 std::cout << s << Ansi::Eol << "\n";
96 };
97
98 // Pads `plain` to the terminal's horizontal centre; callers wrap the
99 // already-centred (and thus already correctly counted) result in ANSI
100 // colour afterwards, so colour codes never throw off the padding math.
101 auto centered = [haveCols](const std::string& plain) -> std::string {
102 const int pad = std::max(0, (haveCols - static_cast<int>(plain.size())) / 2);
103 return std::string(pad, ' ') + plain;
104 };
105
106 std::string modeName;
107 bool canGoBack = false; // only Routing is a runtime sub-state 'b' can actually leave
108 switch (mode)
109 {
110 case DemoMode::Amp: modeName = "Amplifier mode"; break;
111 case DemoMode::SoundscapeFocus: modeName = "Soundscape Focus mode"; break;
112 case DemoMode::Soundscape:
113 if (view == SoundscapeView::Routing) { modeName = "Soundscape Routing sub-mode"; canGoBack = true; }
114 else modeName = "Soundscape Overview mode";
115 break;
116 }
117
118 row("");
119 row(std::string(Ansi::Bold) + Ansi::Red
120 + centered("Terminal too small for " + modeName) + Ansi::Reset);
121 {
122 std::ostringstream o;
123 o << "Needs at least " << neededRows << " rows x " << neededCols << " cols"
124 << " \xe2\x80\x94 have " << haveRows << " x " << haveCols; // U+2014 em dash
125 row(std::string(Ansi::Dim) + centered(o.str()) + Ansi::Reset);
126 }
127 {
128 std::string hint = canGoBack ? "Enlarge the terminal, 'b' to go back, or 'q' to quit"
129 : "Enlarge the terminal, or 'q' to quit";
130 row(std::string(Ansi::Dim) + centered(hint) + Ansi::Reset);
131 }
132 row("");
133}

135static void renderAmpPanel(const AppState& st)
136{
137 const auto sep = makeSep();
138 const auto& a = st.amp;
139
140 auto row = [](const std::string& s) {
141 std::cout << s << Ansi::Eol << "\n";
142 };
143
144 // row 1: title
145 {
146 std::ostringstream o;
147 o << Ansi::Bold << " NanoOcp1 Demo — Amplifier ("
148 << ampTypeStr(a.type) << " " << a.channelCount << "ch)" << Ansi::Reset;
149 row(o.str());
150 }
151 row(sep); // row 2
152
153 // row 3: host / port
154 {
155 std::ostringstream o;
156 o << " Host " << Ansi::Bold << st.address << Ansi::Reset
157 << " Port " << Ansi::Bold << st.port << Ansi::Reset;
158 row(o.str());
159 }
160
161 // row 4: status
162 {
163 std::ostringstream o;
164 renderStatusRow(o, st.ctrlState);
165 row(o.str());
166 }
167
168 // row 5: power
169 {
170 std::ostringstream o;
171 o << " Power ";
172 if (!a.powerKnown)
173 o << Ansi::Dim << "?" << Ansi::Reset;
174 else if (a.powerOn)
175 o << Ansi::Green << "\xe2\x96\xa0 ON" << Ansi::Reset; // U+25A0 filled square
176 else
177 o << Ansi::Red << "\xe2\x96\xa1 OFF" << Ansi::Reset; // U+25A1 open square
178 row(o.str());
179 }
180
181 // rows 6-9: channels (always 4 rows, greyed out if beyond channelCount)
182 for (int i = 0; i < 4; ++i)
183 {
184 std::ostringstream o;
185 const bool active = (i < a.channelCount);
186 if (!active)
187 {
188 o << Ansi::Dim << " Ch" << (i + 1) << " —" << Ansi::Reset;
189 row(o.str());
190 continue;
191 }
192
193 const ChState& ch = a.ch[i];
194 o << " Ch" << (i + 1) << " ";
195
196 // gain
197 o << "Gain ";
198 if (!ch.gainKnown)
199 o << Ansi::Dim << "?" << Ansi::Reset;
200 else
201 o << std::fixed << std::setprecision(1) << std::setw(6) << ch.gainDb << " dB";
202
203 // mute
204 o << " ";
205 if (!ch.muteKnown)
206 o << Ansi::Dim << "mut:?" << Ansi::Reset;
207 else if (ch.muted)
208 o << Ansi::Red << "MUT" << Ansi::Reset;
209 else
210 o << Ansi::Dim << "mut" << Ansi::Reset;
211
212 // ISP GR OVL
213 auto boolLed = [&](bool known, bool active, const char* label) {
214 o << " ";
215 if (!known)
216 o << Ansi::Dim << label << ":?" << Ansi::Reset;
217 else if (active)
218 o << Ansi::Yellow << label << Ansi::Reset;
219 else
220 o << Ansi::Dim << label << Ansi::Reset;
221 };
222 boolLed(ch.ispKnown, ch.isp, "ISP");
223 boolLed(ch.grKnown, ch.gr, "GR");
224 boolLed(ch.ovlKnown, ch.ovl, "OVL");
225
226 // headroom
227 o << " ";
228 if (!ch.hrKnown)
229 o << Ansi::Dim << "Hd:?" << Ansi::Reset;
230 else
231 {
232 o << "Hd:";
233 if (ch.headroomDb < -99.0f)
234 o << Ansi::Dim << " -\xe2\x88\x9e" << Ansi::Reset; // -∞
235 else
236 o << std::fixed << std::setprecision(1) << ch.headroomDb;
237 }
238
239 row(o.str());
240 }
241
242 // The command-reference rows and the Events/log section are rendered
243 // separately — see renderAmpCommands() / renderEventsSection() — always
244 // last, immediately above the input prompt.
245}

262// Renders Soundscape-overview mode's table: a handful of fixed header rows plus
263// one compact row per sound object in the active --soundscape <lo>-<hi> range.
264// Only the data — the command-reference rows and the trailing Events/log
265// section are rendered separately (renderSoOverviewCommands() /
266// renderEventsSection()) so renderPanel() can place --speakerlvls's optional
267// block between the data and them, without this function needing to know
268// whether that block is active.
269static void renderSoOverviewPanel(const AppState& st)
270{
271 const auto sep = makeSep();
272 const auto& ds = st.ds100;
273
274 auto row = [](const std::string& s) {
275 std::cout << s << Ansi::Eol << "\n";
276 };
277
278 // row 1: title
279 {
280 std::ostringstream o;
281 o << Ansi::Bold << " NanoOcp1 Demo — Soundscape Overview, SO "
282 << ds.soundObjectLo;
283 if (ds.soundObjectHi != ds.soundObjectLo)
284 o << "-" << ds.soundObjectHi;
285 o << " (" << ds.objects.size() << " object" << (ds.objects.size() == 1 ? "" : "s") << ")"
286 << Ansi::Reset;
287 row(o.str());
288 }
289 row(sep); // row 2
290
291 // row 3: host / port / device model
292 {
293 std::ostringstream o;
294 o << " Host " << Ansi::Bold << st.address << Ansi::Reset
295 << " Port " << Ansi::Bold << st.port << Ansi::Reset;
296 if (!ds.deviceModel.empty())
297 o << " " << Ansi::Dim << ds.deviceModel << Ansi::Reset;
298 row(o.str());
299 }
300
301 // row 4: status
302 {
303 std::ostringstream o;
304 renderStatusRow(o, st.ctrlState);
305 row(o.str());
306 }
307
308 row(sep); // row 5
309
310 // Channel-strip grid: one column per sound object (an SO-number header row
311 // plus one row per parameter), wrapped into bands of g_soColsPerBand
312 // columns so wide ranges spill onto additional stacked bands instead of
313 // off-screen. Every cell is formatted to a fixed kSoValueWidth-char
314 // plain-text width *before* any ANSI colour is wrapped around it, so
315 // colour codes never throw off std::setw()'s column counting.
316 const int colsPerBand = std::max(1, g_soColsPerBand.load());
317 const int total = static_cast<int>(ds.objects.size());
318
319 auto numCell = [](bool known, double value, int prec) -> std::string {
320 std::ostringstream o;
321 if (!known) { o << std::setw(kSoValueWidth) << "?"; return std::string(Ansi::Dim) + o.str() + Ansi::Reset; }
322 o << std::fixed << std::setprecision(prec) << std::setw(kSoValueWidth) << value;
323 return o.str();
324 };
325 // A full multi-cell meterBar(), not a single glyph, for legibility — full
326 // -120..0 dB range, matching the device's actual metering floor.
327 auto levelBarCell = [](bool known, float dbRaw) -> std::string {
328 if (!known) return std::string(Ansi::Dim) + "[???????]" + Ansi::Reset;
329 return meterBar(dbRaw, -120.0f, 0.0f, kSoLevelBarCells);
330 };
331 // The exact dB value shown underneath the bar.
332 auto levelDbCell = [&numCell](bool known, float dbRaw) { return numCell(known, dbRaw, 1); };
333 auto muteCell = [](bool known, bool muted) -> std::string {
334 std::ostringstream o;
335 o << std::setw(kSoValueWidth) << (!known ? "?" : (muted ? "M" : "-"));
336 const auto s = o.str();
337 if (!known) return std::string(Ansi::Dim) + s + Ansi::Reset;
338 return muted ? (std::string(Ansi::Red) + s + Ansi::Reset) : (std::string(Ansi::Dim) + s + Ansi::Reset);
339 };
340 auto dmCell = [](bool known, int dm) -> std::string {
341 std::ostringstream o;
342 if (!known) { o << std::setw(kSoValueWidth) << "?"; return std::string(Ansi::Dim) + o.str() + Ansi::Reset; }
343 o << std::setw(kSoValueWidth) << dm;
344 return o.str();
345 };
346 auto label = [](const char* text) -> std::string {
347 std::ostringstream o;
348 o << std::left << std::setw(5) << text << std::right << " ";
349 return o.str();
350 };
351 auto soHeaderCell = [](int so) -> std::string {
352 std::ostringstream o;
353 o << std::setw(kSoValueWidth) << so;
354 return std::string(Ansi::Bold) + o.str() + Ansi::Reset;
355 };
356
357 for (int start = 0; start < total; start += colsPerBand)
358 {
359 const int end = std::min(total, start + colsPerBand);
360
361 auto band = [&](const char* rowLabel, const std::function<std::string(const AppState::SoObjState&)>& cell) {
362 std::ostringstream r;
363 r << label(rowLabel);
364 for (int i = start; i < end; ++i)
365 r << cell(ds.objects[i]) << " ";
366 row(r.str());
367 };
368
369 band("SO", [&](const AppState::SoObjState& o) { return soHeaderCell(o.soundObject); });
370 band("Lvl", [&](const AppState::SoObjState& o) { return levelBarCell(o.levelKnown, o.levelDb); });
371 band("dB", [&](const AppState::SoObjState& o) { return levelDbCell(o.levelKnown, o.levelDb); });
372 band("Gain", [&](const AppState::SoObjState& o) { return numCell(o.gainKnown, o.gainDb, 1); });
373 band("Mt", [&](const AppState::SoObjState& o) { return muteCell(o.muteKnown, o.muted); });
374 band("X", [&](const AppState::SoObjState& o) { return numCell(o.posKnown, o.posX, 2); });
375 band("Y", [&](const AppState::SoObjState& o) { return numCell(o.posKnown, o.posY, 2); });
376 band("Z", [&](const AppState::SoObjState& o) { return numCell(o.posKnown, o.posZ, 2); });
377 band("Spr", [&](const AppState::SoObjState& o) { return numCell(o.spreadKnown, o.spread, 2); });
378 band("DM", [&](const AppState::SoObjState& o) { return dmCell(o.dmKnown, o.delayMode); });
379 band("ES", [&](const AppState::SoObjState& o) { return numCell(o.esKnown, o.enspaceDb, 1); });
380
381 if (end < total)
382 row(sep);
383 }
384
385 // The command-reference rows and the Events/log section are rendered
386 // separately — see renderSoOverviewCommands() / renderEventsSection() —
387 // so --speakerlvls's block can be placed between the data and them.
388}

406// Renders the Soundobject-Routing sub-mode table: one *column* per selected
407// sound object (the same column/band scheme as the overview, wrapped onto
408// additional stacked bands when needed) and one row per Function Group
409// (always all 32, dimmed and collapsed when FunctionGroup_Mode says the group
410// is unused). Only the data — the command-reference rows and the trailing
411// Events/log section are rendered separately (renderSoRoutingCommands() /
412// renderEventsSection()) so renderPanel() can place --speakerlvls's optional
413// block between the data and them, without this function needing to know
414// whether that block is active.
415static void renderSoRoutingPanel(const AppState& st)
416{
417 const auto sep = makeSep();
418 const auto& ds = st.ds100;
419
420 auto row = [](const std::string& s) {
421 std::cout << s << Ansi::Eol << "\n";
422 };
423
424 // row 1: title
425 {
426 std::ostringstream o;
427 o << Ansi::Bold << " NanoOcp1 Demo — Soundscape Routing: SO " << ds.routingSoundObjectLo;
428 if (ds.routingSoundObjectHi != ds.routingSoundObjectLo)
429 o << "-" << ds.routingSoundObjectHi;
430 o << " -> Function Groups" << Ansi::Reset;
431 row(o.str());
432 }
433 row(sep); // row 2
434
435 // row 3: host / port / device model
436 {
437 std::ostringstream o;
438 o << " Host " << Ansi::Bold << st.address << Ansi::Reset
439 << " Port " << Ansi::Bold << st.port << Ansi::Reset;
440 if (!ds.deviceModel.empty())
441 o << " " << Ansi::Dim << ds.deviceModel << Ansi::Reset;
442 row(o.str());
443 }
444
445 // row 4: status
446 {
447 std::ostringstream o;
448 renderStatusRow(o, st.ctrlState);
449 row(o.str());
450 }
451
452 row(sep); // row 5
453
454 // Channel-strip grid: one column per selected sound object, one row per
455 // Function Group. As with the overview grid, every cell is formatted to a
456 // fixed plain-text width *before* any ANSI colour is wrapped around it.
457 const int colsPerBand = std::max(1, g_routingColsPerBand.load());
458 const int total = static_cast<int>(ds.routingObjects.size());
459 const int maxFg = static_cast<int>(NanoOcp1::SoundscapeController::sc_MAX_FUNCTION_GROUPS);
460
461 auto gutterLabel = [](const char* text) -> std::string {
462 std::ostringstream o;
463 o << std::left << std::setw(kRoutingLabelGutter) << text << std::right;
464 return o.str();
465 };
466 // "<fg#> <name, truncated>", padded to exactly kRoutingLabelGutter.
467 auto fgLabel = [](int fg, bool nameKnown, const std::string& name) -> std::string {
468 std::ostringstream o;
469 o << std::setw(2) << fg << " ";
470 std::string n = nameKnown ? name : std::string("?");
471 if (n.size() > 17) n = n.substr(0, 17);
472 o << std::left << std::setw(17) << n << std::right;
473 return o.str();
474 };
475 auto soHeaderCell = [](int so) -> std::string {
476 std::ostringstream o;
477 o << std::setw(kRoutingValueWidth) << so;
478 return std::string(Ansi::Bold) + o.str() + Ansi::Reset;
479 };
480 // "<gainDb:6.1f> <mute-glyph>", each sub-field coloured only after its own
481 // fixed-width formatting is already done.
482 auto routingCell = [](bool gainKnown, float gainDb, bool muteKnown, bool muted) -> std::string {
483 std::ostringstream g;
484 if (!gainKnown) g << std::setw(6) << "?";
485 else g << std::fixed << std::setprecision(1) << std::setw(6) << gainDb;
486 const std::string gainStr = gainKnown ? g.str() : (std::string(Ansi::Dim) + g.str() + Ansi::Reset);
487
488 const std::string muteChar = !muteKnown ? "?" : (muted ? "M" : "-");
489 const std::string muteStr = !muteKnown ? (std::string(Ansi::Dim) + muteChar + Ansi::Reset)
490 : muted ? (std::string(Ansi::Red) + muteChar + Ansi::Reset)
491 : (std::string(Ansi::Dim) + muteChar + Ansi::Reset);
492 return gainStr + " " + muteStr;
493 };
494
495 for (int start = 0; start < total; start += colsPerBand)
496 {
497 const int end = std::min(total, start + colsPerBand);
498
499 // SO-number header row
500 {
501 std::ostringstream r;
502 r << gutterLabel("SO");
503 for (int i = start; i < end; ++i)
504 r << " " << soHeaderCell(ds.routingObjects[i].soundObject);
505 row(r.str());
506 }
507
508 // one row per Function Group
509 for (int fg = 1; fg <= maxFg; ++fg)
510 {
511 const auto& info = ds.fg[fg];
512 const bool inUse = info.modeKnown && info.mode != 0;
513
514 std::ostringstream r;
515 if (!inUse)
516 {
517 r << Ansi::Dim << fgLabel(fg, info.nameKnown, info.name);
518 for (int i = start; i < end; ++i)
519 {
520 std::ostringstream c;
521 c << std::setw(kRoutingValueWidth) << "-";
522 r << " " << c.str();
523 }
524 r << Ansi::Reset;
525 }
526 else
527 {
528 r << fgLabel(fg, info.nameKnown, info.name);
529 for (int i = start; i < end; ++i)
530 {
531 const auto& cell = ds.routingObjects[i].fg[fg];
532 r << " " << routingCell(cell.gainKnown, cell.gainDb, cell.muteKnown, cell.muted);
533 }
534 }
535 row(r.str());
536 }
537
538 if (end < total)
539 row(sep);
540 }

543 // separately — see renderSoRoutingCommands() / renderEventsSection() —
544 // so --speakerlvls's block can be placed between the data and them.
545}

560// Renders the Soundscape-focus monitor panel: a handful of fixed rows describing
561// the focused parameter plus a scrolling, timestamped log that is sized to fill
562// the rest of the terminal (see redrawLoop()'s resize handling). Emits exactly
563// kFocusFixedLines + logCapacity newline-terminated rows.
564static void renderFocusPanel(const AppState& st, const std::vector<std::string>& log, int logCapacity)
565{
566 const auto sep = makeSep();
567 const auto& f = st.focus;
568
569 auto row = [](const std::string& s) {
570 std::cout << s << Ansi::Eol << "\n";
571 };
572
573 // row 1: title
574 {
575 std::ostringstream o;
576 o << Ansi::Bold << " NanoOcp1 Demo — Soundscape Focus: " << f.paramName << Ansi::Reset
577 << " Addr(" << f.addr.pri;
578 if (f.addr.sec != 0)
579 o << ", " << f.addr.sec;
580 o << ")";
581 row(o.str());
582 }
583 row(sep); // row 2
584
585 // row 3: host / port / device model
586 {
587 std::ostringstream o;
588 o << " Host " << Ansi::Bold << st.address << Ansi::Reset
589 << " Port " << Ansi::Bold << st.port << Ansi::Reset;
590 if (!st.ds100.deviceModel.empty())
591 o << " " << Ansi::Dim << st.ds100.deviceModel << Ansi::Reset;
592 row(o.str());
593 }
594
595 // row 4: status
596 {
597 std::ostringstream o;
598 renderStatusRow(o, st.ctrlState);
599 row(o.str());
600 }
601

649// Renders the optional --speakerlvls loudspeaker level-meter block, appended
650// below whichever Soundscape view (overview or routing) is currently shown.
651// Only outputs with a non-zero (assigned) Positioning_SpeakerPosition are
652// shown, laid out as one column per output wrapped into bands exactly like
653// the overview/routing grids — just with a short *vertical* bar (several
654// stacked single-character rows) instead of a horizontal one, since these
655// columns are narrower. Emits exactly
656// soSpeakerBlockLines(assignedCount, g_spkColsPerBand) rows.
657static void renderSpeakerLevelsBlock(const AppState::SoState& ds)
658{
659 const auto sep = makeSep();
660 auto row = [](const std::string& s) {
661 std::cout << s << Ansi::Eol << "\n";
662 };
663
664 row(sep);
665
666 std::vector<const AppState::SpeakerObjState*> assigned;
667 for (const auto& s : ds.speakerObjects)
668 if (s.hasSpeaker) assigned.push_back(&s);
669
670 {
671 std::ostringstream o;
672 o << Ansi::Bold << " Loudspeaker Levels" << Ansi::Reset;
673 if (!assigned.empty())
674 o << " (" << assigned.size() << " output" << (assigned.size() == 1 ? "" : "s") << ")";
675 else
676 o << Ansi::Dim << " (no outputs with an assigned position yet)" << Ansi::Reset;
677 row(o.str());
678 }
679
680 const int colsPerBand = std::max(1, g_spkColsPerBand.load());
681 const int total = static_cast<int>(assigned.size());
682
683 auto label = [](const char* text) -> std::string {
684 std::ostringstream o;
685 o << std::left << std::setw(kSpkLabelGutter - 1) << text << std::right << " ";
686 return o.str();
687 };
688 auto headerCell = [](int ch) -> std::string {
689 std::ostringstream o;
690 o << std::setw(kSpkValueWidth) << ch;
691 return std::string(Ansi::Bold) + o.str() + Ansi::Reset;
692 };
693 auto dbCell = [](bool known, float dB) -> std::string {
694 std::ostringstream o;
695 if (!known) { o << std::setw(kSpkValueWidth) << "?"; return std::string(Ansi::Dim) + o.str() + Ansi::Reset; }
696 o << std::fixed << std::setprecision(1) << std::setw(kSpkValueWidth) << dB;
697 return o.str();
698 };
699 // The bar glyph is always exactly 1 visible character; pad it to
700 // kSpkValueWidth with plain (uncoloured) leading spaces so the colour
701 // codes inside the glyph never factor into the width.
702 auto barCellPad = [](const std::string& glyph) -> std::string {
703 return std::string(kSpkValueWidth - 1, ' ') + glyph;
704 };
705
706 // Runs at least once even when `total == 0`, so the structural rows
707 // (header/bar/dB) still appear — with empty column tails — before any
708 // Positioning_SpeakerPosition values have arrived, matching
709 // soSpeakerBlockLines()'s own std::max(1, assignedCount).
710 for (int start = 0; start < std::max(1, total); start += colsPerBand)
711 {
712 const int end = std::min(total, start + colsPerBand);
713
714 {
715 std::ostringstream r;
716 r << label("Out");
717 for (int i = start; i < end; ++i)
718 r << headerCell(assigned[static_cast<std::size_t>(i)]->outputChannel) << " ";
719 row(r.str());
720 }
721
722 // The bar is clipped to a -30..0 dB window for sensitivity near the
723 // top of range; the dB row below it still shows the true value down
724 // to the device's full -120 dB floor (see dbCell()).
725 std::vector<std::vector<std::string>> bars;
726 for (int i = start; i < end; ++i)
727 {
728 const auto* s = assigned[static_cast<std::size_t>(i)];
729 bars.push_back(verticalBarGlyphs(s->levelKnown, s->levelDb, -30.0f, 0.0f, kSpkBarSegments));
730 }
731
732 for (int seg = 0; seg < kSpkBarSegments; ++seg)
733 {
734 std::ostringstream r;
735 r << label("");
736 for (int i = start; i < end; ++i)
737 r << barCellPad(bars[static_cast<std::size_t>(i - start)][static_cast<std::size_t>(seg)]) << " ";
738 row(r.str());
739 }
740
741 {
742 std::ostringstream r;
743 r << label("dB");
744 for (int i = start; i < end; ++i)
745 r << dbCell(assigned[static_cast<std::size_t>(i)]->levelKnown,
746 assigned[static_cast<std::size_t>(i)]->levelDb) << " ";
747 row(r.str());
748 }
749
750 if (end < total)
751 row(sep);
752 }
753}

755static void renderPanel()
756{
757 AppState state;
758 std::vector<std::string> log;
759 {
760 std::lock_guard<std::mutex> lk(g_stateMutex);
761 state = g_state;
762 log.assign(g_log.begin(), g_log.end());
763 }
764
765 if (!g_sizeFits.load())
766 {
767 renderTooSmallWarning(state.mode, state.ds100.view,
768 g_neededRows.load(), g_neededCols.load(),
769 g_termRows.load(), g_termCols.load());
770 return;
771 }
772
773 // Section order (Amp / Soundscape overview / Soundscape routing): data,
774 // then any optional trailing block (--speakerlvls), then the command
775 // legend, then Events/log — always the very last section, immediately
776 // above the input prompt. Each renderer below only ever emits its own
777 // data; composing that order is renderPanel()'s job alone, so none of
778 // them need to know or care whether --speakerlvls is active.
779 switch (state.mode)
780 {
781 case DemoMode::Amp:
782 renderAmpPanel(state);
783 renderAmpCommands();
784 renderEventsSection(log, kLogLines);
785 break;
786 case DemoMode::Soundscape:
787 if (state.ds100.view == SoundscapeView::Routing)
788 renderSoRoutingPanel(state);
789 else
790 renderSoOverviewPanel(state);
791 if (state.ds100.speakerLvlsEnabled)
792 renderSpeakerLevelsBlock(state.ds100);
793 if (state.ds100.view == SoundscapeView::Routing)
794 renderSoRoutingCommands();
795 else
796 renderSoOverviewCommands();
797 renderEventsSection(log, kLogLines);
798 break;
799 case DemoMode::SoundscapeFocus:
800 renderFocusPanel(state, log, g_logCapacity.load());
801 break;
802 }
803}

825// ── Background redraw thread ──────────────────────────────────────────────────
826static void redrawLoop()
827{
828 while (!g_quit)
829 {
830 DemoMode mode;
831 SoundscapeView view;
832 int overviewRangeCount, routingRangeCount;
833 bool speakerLvlsEnabled;
834 int assignedSpeakerCount;
835 {
836 std::lock_guard<std::mutex> lk(g_stateMutex);
837 mode = g_state.mode;
838 view = g_state.ds100.view;
839 overviewRangeCount = g_state.ds100.soundObjectHi - g_state.ds100.soundObjectLo + 1;
840 routingRangeCount = g_state.ds100.routingSoundObjectHi - g_state.ds100.routingSoundObjectLo + 1;
841 speakerLvlsEnabled = g_state.ds100.speakerLvlsEnabled;
842 assignedSpeakerCount = static_cast<int>(std::count_if(
843 g_state.ds100.speakerObjects.begin(), g_state.ds100.speakerObjects.end(),
844 [](const AppState::SpeakerObjState& s) { return s.hasSpeaker; }));
845 }
846
847 // Every mode's row count — and, for Soundscape overview/routing, how
848 // many sound-object columns fit per band — depends on the current
849 // terminal size, which can change at runtime (a live resize) just as
850 // easily as it can start out too small. computePanelFit() also tells
851 // us whether the current size is too small to use at all, in which
852 // case the panel is swapped for a short warning (see renderPanel()).
853 int rows, cols;
854 if (getTerminalSize(rows, cols))
855 {
856 g_termRows = rows;
857 g_termCols = cols;
858 }
859 else
860 {
861 rows = g_termRows.load();
862 cols = g_termCols.load();
863 }
864
865 const PanelFit fit = computePanelFit(mode, view, overviewRangeCount, routingRangeCount,
866 speakerLvlsEnabled, assignedSpeakerCount, rows, cols);
867 g_sizeFits = fit.fits;
868 g_neededRows = fit.neededRows;
869 g_neededCols = fit.neededCols;
870 if (mode == DemoMode::Soundscape)
871 {
872 if (view == SoundscapeView::Routing) g_routingColsPerBand = fit.colsPerBand;
873 else g_soColsPerBand = fit.colsPerBand;
874 if (speakerLvlsEnabled) g_spkColsPerBand = fit.spkColsPerBand;
875 }
876
877 int panelLines;
878 if (!fit.fits)
879 {
880 panelLines = kWarningPanelLines;
881 }
882 else if (mode == DemoMode::SoundscapeFocus)
883 {
884 const int logLines = std::max(kFocusMinLogLines, rows - kFocusFixedLines - 1);
885 panelLines = kFocusFixedLines + logLines;
886 g_logCapacity = logLines;
887 }
888 else if (mode == DemoMode::Soundscape)
889 {
890 panelLines = fit.neededRows - 1; // neededRows includes the prompt row; panelLines doesn't
891 }
892 else // Amp
893 {
894 panelLines = kPanelLines;
895 }
896
897 if (panelLines != g_panelLines.load())
898 {
899 resetCanvas(panelLines);
900 g_panelLines = panelLines;
901 g_needsRedraw = true;
902 }
903
904 if (g_needsRedraw.exchange(false))
905 {
906 std::lock_guard<std::mutex> lk(g_ioMutex);
907 std::cout << Ansi::Save << Ansi::Home;
908 renderPanel();
909 std::cout << Ansi::Rest;
910 std::cout.flush();
911 }
912 std::this_thread::sleep_for(std::chrono::milliseconds(120));
913 }
914}
