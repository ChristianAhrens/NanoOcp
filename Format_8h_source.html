34static std::string fillBar(float val, float lo, float hi, int width = 16)
35{
36 const float clamped = std::max(lo, std::min(hi, val));
37 const float frac = (hi > lo) ? (clamped - lo) / (hi - lo) : 0.0f;
38 const int n = static_cast<int>(std::round(frac * static_cast<float>(width)));
39 std::string bar = "[";
40 for (int i = 0; i < width; ++i)
41 bar += (i < n) ? "\xe2\x96\x88" : "\xe2\x96\x91"; // U+2588 full / U+2591 light
42 bar += "]";
43 return bar;
44}

46// Colour-segmented VU-meter-style bar (green/yellow/red zones), redrawn on every
47// live value update from the device — used for signal-level and channel-gain
48// readouts, as opposed to fillBar()'s plain single-colour rendering for
49// positional/target controls (spread, position, delay mode).
50static std::string meterBar(float val, float lo, float hi, int width = 16)
51{
52 const float clamped = std::max(lo, std::min(hi, val));
53 const float frac = (hi > lo) ? (clamped - lo) / (hi - lo) : 0.0f;
54 const int n = static_cast<int>(std::round(frac * static_cast<float>(width)));
55 std::string bar = "[";
56 for (int i = 0; i < width; ++i)
57 {
58 if (i >= n)
59 {
60 bar += "\xe2\x96\x91"; // U+2591 light — unlit cell
61 continue;
62 }
63 const float cellFrac = static_cast<float>(i + 1) / static_cast<float>(width);
64 const char* color = (cellFrac > 0.9f) ? Ansi::Red
65 : (cellFrac > 0.75f) ? Ansi::Yellow
66 : Ansi::Green;
67 bar += std::string(color) + "\xe2\x96\x88" + Ansi::Reset; // U+2588 full, zone-coloured
68 }
69 bar += "]";
70 return bar;
71}
